//
//  Diag2gUITests.swift — temporary harness for pass 2g (2026-09-15). Not committed.
//  Attach mode (the app is started by `devicectl … --console`), as passes 2e/2f.
//
//  One session, Stargate Extended only (the owner tests the rest by hand):
//    play 30 s; 4 × Right (+30 s); 20 s;
//    pause (Play/Pause), 3 s; then three drags with no landing in between, so the target and the
//    still above the bar move each time — Page Up = +0.08 of the width (≈ +156 s), Page Down =
//    −0.04 (≈ −78 s) — a screenshot mid-drag and one after each lift;
//    click (land), screenshot +3 s, 20 s of playback;
//    pause, 3 s, one drag, screenshot mid-drag, Menu (cancel; stays paused at the drag's start);
//    5 × Left and 5 × Right clicks (frame step), a screenshot each;
//    Menu (exit).
//

import XCTest

final class Diag2gUITests: XCTestCase {
    private var app: XCUIApplication!
    private let remote = XCUIRemote.shared
    private let headerOrder = ["tab.Movies", "tab.TV Shows", "tab.Videos", "sort"]

    override func setUp() { continueAfterFailure = true }

    private func shot(_ name: String) {
        let a = XCTAttachment(screenshot: XCUIScreen.main.screenshot()); a.name = name; a.lifetime = .keepAlways; add(a)
    }
    private func note(_ t: String) { let a = XCTAttachment(string: t); a.name = "note"; a.lifetime = .keepAlways; add(a); NSLog("[p2g] %@", t) }
    private func press(_ b: XCUIRemote.Button, _ n: Int = 1, wait: UInt32 = 1) { for _ in 0..<n { remote.press(b); sleep(wait) } }
    private func focusedIds() -> [String] {
        app.descendants(matching: .any).matching(NSPredicate(format: "hasFocus == YES")).allElementsBoundByIndex.map { $0.identifier.isEmpty ? "(\($0.label))" : $0.identifier }
    }
    private func isFocused(_ id: String) -> Bool { focusedIds().contains(id) }
    private func isFocused(prefix: String) -> Bool { focusedIds().contains { $0.hasPrefix(prefix) } }
    @discardableResult private func moveUntilFocused(_ id: String, pressing b: XCUIRemote.Button, limit: Int = 8) -> Bool {
        for _ in 0..<limit { if isFocused(id) { return true }; press(b) }
        return isFocused(id)
    }
    private func focusHeader() { for _ in 0..<8 { if isFocused(prefix: "tab.") || isFocused("sort") { return }; press(.up) } }
    private func focusHeaderItem(_ id: String) {
        focusHeader(); guard let target = headerOrder.firstIndex(of: id) else { return }
        for _ in 0..<8 { if isFocused(id) { return }; let cur = headerOrder.firstIndex { isFocused($0) } ?? 0; press(cur < target ? .right : .left) }
    }
    private func selectTab(_ name: String) { focusHeaderItem("tab.\(name)"); press(.select, wait: 2) }
    private func openPoster(_ id: String, tab: String) {
        selectTab(tab); press(.down); if !isFocused(prefix: "poster.") { press(.down) }
        _ = moveUntilFocused(id, pressing: .right, limit: 6) || moveUntilFocused(id, pressing: .left, limit: 6)
        press(.select, wait: 3)
        XCTAssertTrue(app.staticTexts["detail.title"].waitForExistence(timeout: 20), "detail did not open; focus \(focusedIds())")
        sleep(3)   // the detail screen's one thumbs index request lands in here
    }
    private func pressPlay() { if !isFocused("play") { moveUntilFocused("play", pressing: .up, limit: 6) }; press(.select, wait: 2) }
    private func stamp() -> String { let f = DateFormatter(); f.dateFormat = "HH:mm:ss.SSS"; return f.string(from: Date()) }
    private func pickEdition(_ name: String) {
        XCTAssertTrue(app.buttons["pick.\(name)"].waitForExistence(timeout: 5), "edition picker did not open")
        _ = moveUntilFocused("pick.\(name)", pressing: .up, limit: 3) || moveUntilFocused("pick.\(name)", pressing: .down, limit: 3)
        press(.select, wait: 2)
    }
    private func attach() {
        app = XCUIApplication(); app.activate()
        note("attached at \(stamp())")
        XCTAssertTrue(app.buttons["tab.Movies"].waitForExistence(timeout: 60), "library did not load")
        sleep(3)
    }

    func tStargateThumbs() {
        let tag = "stargate-thumbs"
        attach()
        openPoster("poster.Stargate", tab: "Movies")
        note("\(tag): detail open (thumbs index requested) at \(stamp())")
        shot("p2g-\(tag)-00-detail")
        pressPlay(); pickEdition("Extended")
        XCTAssertTrue(app.otherElements["player"].waitForExistence(timeout: 20), "player did not appear; focus \(focusedIds())")
        note("\(tag): player up at \(stamp())")
        sleep(30)
        for n in 1...4 { note("\(tag): skip +30 \(n) at \(stamp())"); remote.press(.right); usleep(1_500_000) }
        sleep(20)

        note("\(tag): pause (Play/Pause) at \(stamp())")
        remote.press(.playPause)
        sleep(3)
        shot("p2g-\(tag)-01-paused-before-drag")

        // Three drags with no landing between them: each moves the target on from the last one.
        for (i, forward) in [true, true, false].enumerated() {
            note("\(tag): drag \(i + 1) \(forward ? "forward" : "back") at \(stamp())")
            remote.press(forward ? .pageUp : .pageDown)
            usleep(600_000)
            shot(String(format: "p2g-\(tag)-%02d-drag%d-mid-a", i * 2 + 2, i + 1))
            shot(String(format: "p2g-\(tag)-%02d-drag%d-mid-b", i * 2 + 3, i + 1))
            sleep(2)
            shot(String(format: "p2g-\(tag)-%02d-drag%d-lifted", i * 2 + 4, i + 1))
            sleep(1)
        }

        note("\(tag): land (click) at \(stamp())")
        remote.press(.select)
        sleep(3)
        shot("p2g-\(tag)-20-land+3s")
        sleep(20)

        note("\(tag): pause before the cancel drag at \(stamp())")
        remote.press(.playPause)
        sleep(3)
        note("\(tag): cancel drag (forward) at \(stamp())")
        remote.press(.pageUp)
        usleep(900_000)
        shot("p2g-\(tag)-21-cancel-drag-mid")
        sleep(2)
        shot("p2g-\(tag)-22-cancel-drag-lifted")
        note("\(tag): menu (cancel) at \(stamp())")
        remote.press(.menu)
        sleep(3)
        shot("p2g-\(tag)-23-after-cancel+3s")
        XCTAssertTrue(app.otherElements["player"].exists, "Menu cancelled the scrub but also left the player")

        for n in 1...10 {
            let left = n <= 5
            note("\(tag): frame click \(n) \(left ? "L" : "R") at \(stamp())")
            remote.press(left ? .left : .right)
            sleep(3)
            shot(String(format: "p2g-\(tag)-fs-%02d-\(left ? "L" : "R")", n))
            usleep(1_200_000)
        }

        note("\(tag): exit at \(stamp())")
        press(.menu, wait: 3)
        XCTAssertFalse(app.otherElements["player"].exists, "player still up after the final Menu")
    }
}
