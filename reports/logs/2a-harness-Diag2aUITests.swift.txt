//
//  Diag2aUITests.swift — temporary harness for pass 2a (2026-09-15). Not committed.
//  Attach mode (the app is started by `devicectl … --console` with the run's VLCParams, tracer on), from pass 1k.
//  XCUIRemote cannot touch the Siri Remote's surface, so a drag is Page Up (forward) / Page Down (back), which the
//  uncommitted harness hook in PlayerHost turns into a 2 s scripted drag through the model's scrub calls.
//  Scrub run: play 30 s, 4 × Right (+30 s skips) 1.5 s apart, 20 s, [pause, 5 s], drag, screenshot mid-drag (+1 s)
//  and after lift (+3 s), hold to +8 s, Select (land) or Menu (cancel), screenshots at +3 s and +30 s, Menu.
//  Control run: play 30 s, Right, Left, pause, 3 × Left and 3 × Right clicks, Play, Up + Select (audio panel),
//  Menu (closes it), Menu (exits).
//

import XCTest

final class Diag2aUITests: XCTestCase {
    private var app: XCUIApplication!
    private let remote = XCUIRemote.shared
    private let headerOrder = ["tab.Movies", "tab.TV Shows", "tab.Videos", "sort"]

    override func setUp() { continueAfterFailure = true }

    private func shot(_ name: String) {
        let a = XCTAttachment(screenshot: XCUIScreen.main.screenshot()); a.name = name; a.lifetime = .keepAlways; add(a)
    }
    private func note(_ t: String) { let a = XCTAttachment(string: t); a.name = "note"; a.lifetime = .keepAlways; add(a); NSLog("[p2a] %@", t) }
    private func press(_ b: XCUIRemote.Button, _ n: Int = 1, wait: UInt32 = 1) { for _ in 0..<n { remote.press(b); sleep(wait) } }
    private func focusedIds() -> [String] {
        app.descendants(matching: .any).matching(NSPredicate(format: "hasFocus == YES")).allElementsBoundByIndex.map { $0.identifier.isEmpty ? "(\($0.label))" : $0.identifier }
    }
    private func isFocused(_ id: String) -> Bool { focusedIds().contains(id) }
    private func isFocused(prefix: String) -> Bool { focusedIds().contains { $0.hasPrefix(prefix) } }
    @discardableResult private func moveUntilFocused(_ id: String, pressing b: XCUIRemote.Button, limit: Int = 8) -> Bool {
        for _ in 0..<limit { if isFocused(id) { return true }; press(b) }
        return isFocused(id)
    }
    private func focusHeader() { for _ in 0..<8 { if isFocused(prefix: "tab.") || isFocused("sort") { return }; press(.up) } }
    private func focusHeaderItem(_ id: String) {
        focusHeader(); guard let target = headerOrder.firstIndex(of: id) else { return }
        for _ in 0..<8 { if isFocused(id) { return }; let cur = headerOrder.firstIndex { isFocused($0) } ?? 0; press(cur < target ? .right : .left) }
    }
    private func selectTab(_ name: String) { focusHeaderItem("tab.\(name)"); press(.select, wait: 2) }
    private func openPoster(_ id: String, tab: String) {
        selectTab(tab); press(.down); if !isFocused(prefix: "poster.") { press(.down) }
        _ = moveUntilFocused(id, pressing: .right, limit: 6) || moveUntilFocused(id, pressing: .left, limit: 6)
        press(.select, wait: 3)
        XCTAssertTrue(app.staticTexts["detail.title"].waitForExistence(timeout: 20), "detail did not open; focus \(focusedIds())")
        sleep(2)
    }
    private func pressPlay() { if !isFocused("play") { moveUntilFocused("play", pressing: .up, limit: 6) }; press(.select, wait: 2) }
    private func stamp() -> String { let f = DateFormatter(); f.dateFormat = "HH:mm:ss.SSS"; return f.string(from: Date()) }
    private func openStargate() {
        openPoster("poster.Stargate", tab: "Movies"); pressPlay()
        XCTAssertTrue(app.buttons["pick.Extended"].waitForExistence(timeout: 5)); moveUntilFocused("pick.Extended", pressing: .up, limit: 3); press(.select, wait: 2)
    }
    private func openEpisode(show: String, season: Int, episode: Int) {
        openPoster("poster.\(show)", tab: "TV Shows")
        XCTAssertTrue(app.buttons["season.\(season)"].waitForExistence(timeout: 30), "seasons did not load"); sleep(2)
        if !app.buttons["episode.\(season).\(episode)"].exists {
            moveUntilFocused("season.\(season)", pressing: .right, limit: 4); press(.select, wait: 2)
        }
        XCTAssertTrue(moveUntilFocused("episode.\(season).\(episode)", pressing: .down, limit: 6), "episode not focused; focus \(focusedIds())")
        press(.select, wait: 2)
    }
    private func open(_ film: String) {
        switch film {
        case "stargate": openStargate()
        case "ww": openPoster("poster.Wonder Woman", tab: "Movies"); pressPlay()
        case "divergent": openPoster("poster.Divergent", tab: "Movies"); pressPlay()
        default: openEpisode(show: "The Magicians", season: 1, episode: 1)
        }
    }
    private func attach() {
        app = XCUIApplication(); app.activate()
        note("attached at \(stamp())")
        XCTAssertTrue(app.buttons["tab.Movies"].waitForExistence(timeout: 60), "library did not load")
        sleep(3)
    }
    private func playerUp(_ tag: String) {
        XCTAssertTrue(app.otherElements["player"].waitForExistence(timeout: 20), "player did not appear; focus \(focusedIds())")
        note("\(tag): player up at \(stamp())")
    }

    /// paused: pause before the drag. forward: Page Up (+) or Page Down (−). cancel: Menu instead of Select.
    private func scrub(_ film: String, paused: Bool, forward: Bool, cancel: Bool) {
        let tag = "\(film)-\(paused ? "pause" : "play")\(cancel ? "cancel" : (forward ? "fwd" : "back"))"
        attach(); open(film); playerUp(tag)
        sleep(30)
        for n in 1...4 { note("\(tag): skip +30 \(n) at \(stamp())"); remote.press(.right); usleep(1_500_000) }
        sleep(20)
        if paused { note("\(tag): pause at \(stamp())"); press(.select, wait: 5) }
        note("\(tag): drag \(forward ? "forward" : "back") at \(stamp())")
        remote.press(forward ? .pageUp : .pageDown)
        usleep(1_000_000)
        shot("p2a-\(tag)-01-mid-drag")
        sleep(2)
        shot("p2a-\(tag)-02-lifted")
        sleep(5)
        note("\(tag): \(cancel ? "menu (cancel)" : "select (land)") at \(stamp())")
        remote.press(cancel ? .menu : .select)
        sleep(3)
        shot("p2a-\(tag)-03-after+3s")
        sleep(26)
        press(.up, wait: 1)
        shot("p2a-\(tag)-04-after+30s")
        note("\(tag): after+30s shot at \(stamp()), player exists \(app.otherElements["player"].exists)")
        press(.down, wait: 1)
        press(.menu, wait: 3)
        XCTAssertFalse(app.otherElements["player"].exists, "player still up after the final Menu")
    }

    /// Step 4: skips while playing, frame steps while paused, Menu closes a panel before it exits.
    private func control(_ film: String) {
        let tag = "\(film)-control"
        attach(); open(film); playerUp(tag)
        sleep(30)
        note("\(tag): right (+30) at \(stamp())"); press(.right, wait: 3)
        note("\(tag): left (−10) at \(stamp())"); press(.left, wait: 3)
        note("\(tag): pause at \(stamp())"); press(.select, wait: 3)
        for n in 1...3 { note("\(tag): click \(n) L at \(stamp())"); press(.left, wait: 3) }
        for n in 4...6 { note("\(tag): click \(n) R at \(stamp())"); press(.right, wait: 3) }
        shot("p2a-\(tag)-01-after-steps")
        note("\(tag): play at \(stamp())"); press(.select, wait: 5)
        press(.up, wait: 1)
        note("\(tag): open audio panel at \(stamp())"); press(.select, wait: 2)
        XCTAssertTrue(app.staticTexts["AUDIO"].waitForExistence(timeout: 5), "the audio panel did not open")
        shot("p2a-\(tag)-02-panel")
        note("\(tag): menu 1 at \(stamp())"); press(.menu, wait: 2)
        XCTAssertTrue(app.otherElements["player"].exists, "Menu with the panel open closed the player")
        shot("p2a-\(tag)-03-panel-closed")
        note("\(tag): menu 2 at \(stamp())"); press(.menu, wait: 3)
        XCTAssertFalse(app.otherElements["player"].exists, "the second Menu did not exit the player")
    }

    func t0_Probe() {
        let tag = "magicians-probe"
        attach(); open("magicians"); playerUp(tag)
        sleep(15)
        note("\(tag): drag forward at \(stamp())"); remote.press(.pageUp)
        usleep(1_000_000); shot("p2a-\(tag)-01-mid-drag")
        sleep(4); shot("p2a-\(tag)-02-lifted")
        note("\(tag): select (land) at \(stamp())"); remote.press(.select)
        sleep(5); shot("p2a-\(tag)-03-after+5s")
        press(.menu, wait: 3)
    }

    func tStargatePlayFwd() { scrub("stargate", paused: false, forward: true, cancel: false) }
    func tStargatePlayBack() { scrub("stargate", paused: false, forward: false, cancel: false) }
    func tStargatePauseFwd() { scrub("stargate", paused: true, forward: true, cancel: false) }
    func tStargatePauseBack() { scrub("stargate", paused: true, forward: false, cancel: false) }
    func tStargatePlayCancel() { scrub("stargate", paused: false, forward: true, cancel: true) }
    func tStargatePauseCancel() { scrub("stargate", paused: true, forward: true, cancel: true) }
    func tStargateControl() { control("stargate") }
    func tWonderWomanPlayFwd() { scrub("ww", paused: false, forward: true, cancel: false) }
    func tWonderWomanPlayBack() { scrub("ww", paused: false, forward: false, cancel: false) }
    func tWonderWomanPauseFwd() { scrub("ww", paused: true, forward: true, cancel: false) }
    func tWonderWomanPauseBack() { scrub("ww", paused: true, forward: false, cancel: false) }
    func tWonderWomanPlayCancel() { scrub("ww", paused: false, forward: true, cancel: true) }
    func tWonderWomanPauseCancel() { scrub("ww", paused: true, forward: true, cancel: true) }
    func tWonderWomanControl() { control("ww") }
    func tDivergentPlayFwd() { scrub("divergent", paused: false, forward: true, cancel: false) }
    func tDivergentPlayBack() { scrub("divergent", paused: false, forward: false, cancel: false) }
    func tDivergentPauseFwd() { scrub("divergent", paused: true, forward: true, cancel: false) }
    func tDivergentPauseBack() { scrub("divergent", paused: true, forward: false, cancel: false) }
    func tDivergentPlayCancel() { scrub("divergent", paused: false, forward: true, cancel: true) }
    func tDivergentPauseCancel() { scrub("divergent", paused: true, forward: true, cancel: true) }
    func tDivergentControl() { control("divergent") }
    func tMagiciansPlayFwd() { scrub("magicians", paused: false, forward: true, cancel: false) }
    func tMagiciansPlayBack() { scrub("magicians", paused: false, forward: false, cancel: false) }
    func tMagiciansPauseFwd() { scrub("magicians", paused: true, forward: true, cancel: false) }
    func tMagiciansPauseBack() { scrub("magicians", paused: true, forward: false, cancel: false) }
    func tMagiciansPlayCancel() { scrub("magicians", paused: false, forward: true, cancel: true) }
    func tMagiciansPauseCancel() { scrub("magicians", paused: true, forward: true, cancel: true) }
    func tMagiciansControl() { control("magicians") }
}
