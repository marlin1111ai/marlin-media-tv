//
//  Diag2dUITests.swift — temporary harness for pass 2d (2026-09-15). Not committed.
//  Attach mode (the app is started by `devicectl … --console` with the run's VLCParams, tracer on), from pass 2c.
//  XCUIRemote cannot touch the Siri Remote's surface: Page Up (forward) / Page Down (back) play the uncommitted
//  harness hook's 2 s scripted drag through the same model calls as the pan recognizer.
//  Scrub session (paused): play 30 s, 4 × Right (+30 s), 20 s, then six scrubs, pausing before each one that follows a
//  landing — fwd × Play/Pause, back × click, fwd × click, back × Play/Pause, fwd × Menu, back × Menu. Each: drag,
//  screenshot mid-drag (+1 s), the ending at +5 s, screenshot 3 s later, 29 s more after a landing (7 s after Menu).
//  Control session: play 30 s; while playing a forward then a back drag (screenshots); +30 s, −10 s; pause;
//  5 × Left and 5 × Right clicks ~4.2 s apart with a screenshot 3 s after each; Play; Menu.
//

import XCTest

final class Diag2dUITests: XCTestCase {
    private var app: XCUIApplication!
    private let remote = XCUIRemote.shared
    private let headerOrder = ["tab.Movies", "tab.TV Shows", "tab.Videos", "sort"]

    override func setUp() { continueAfterFailure = true }

    private func shot(_ name: String) {
        let a = XCTAttachment(screenshot: XCUIScreen.main.screenshot()); a.name = name; a.lifetime = .keepAlways; add(a)
    }
    private func note(_ t: String) { let a = XCTAttachment(string: t); a.name = "note"; a.lifetime = .keepAlways; add(a); NSLog("[p2d] %@", t) }
    private func press(_ b: XCUIRemote.Button, _ n: Int = 1, wait: UInt32 = 1) { for _ in 0..<n { remote.press(b); sleep(wait) } }
    private func focusedIds() -> [String] {
        app.descendants(matching: .any).matching(NSPredicate(format: "hasFocus == YES")).allElementsBoundByIndex.map { $0.identifier.isEmpty ? "(\($0.label))" : $0.identifier }
    }
    private func isFocused(_ id: String) -> Bool { focusedIds().contains(id) }
    private func isFocused(prefix: String) -> Bool { focusedIds().contains { $0.hasPrefix(prefix) } }
    @discardableResult private func moveUntilFocused(_ id: String, pressing b: XCUIRemote.Button, limit: Int = 8) -> Bool {
        for _ in 0..<limit { if isFocused(id) { return true }; press(b) }
        return isFocused(id)
    }
    private func focusHeader() { for _ in 0..<8 { if isFocused(prefix: "tab.") || isFocused("sort") { return }; press(.up) } }
    private func focusHeaderItem(_ id: String) {
        focusHeader(); guard let target = headerOrder.firstIndex(of: id) else { return }
        for _ in 0..<8 { if isFocused(id) { return }; let cur = headerOrder.firstIndex { isFocused($0) } ?? 0; press(cur < target ? .right : .left) }
    }
    private func selectTab(_ name: String) { focusHeaderItem("tab.\(name)"); press(.select, wait: 2) }
    private func openPoster(_ id: String, tab: String) {
        selectTab(tab); press(.down); if !isFocused(prefix: "poster.") { press(.down) }
        _ = moveUntilFocused(id, pressing: .right, limit: 6) || moveUntilFocused(id, pressing: .left, limit: 6)
        press(.select, wait: 3)
        XCTAssertTrue(app.staticTexts["detail.title"].waitForExistence(timeout: 20), "detail did not open; focus \(focusedIds())")
        sleep(2)
    }
    private func pressPlay() { if !isFocused("play") { moveUntilFocused("play", pressing: .up, limit: 6) }; press(.select, wait: 2) }
    private func stamp() -> String { let f = DateFormatter(); f.dateFormat = "HH:mm:ss.SSS"; return f.string(from: Date()) }
    private func openStargate() {
        openPoster("poster.Stargate", tab: "Movies"); pressPlay()
        XCTAssertTrue(app.buttons["pick.Extended"].waitForExistence(timeout: 5)); moveUntilFocused("pick.Extended", pressing: .up, limit: 3); press(.select, wait: 2)
    }
    private func openEpisode(show: String, season: Int, episode: Int) {
        openPoster("poster.\(show)", tab: "TV Shows")
        XCTAssertTrue(app.buttons["season.\(season)"].waitForExistence(timeout: 30), "seasons did not load"); sleep(2)
        if !app.buttons["episode.\(season).\(episode)"].exists {
            moveUntilFocused("season.\(season)", pressing: .right, limit: 4); press(.select, wait: 2)
        }
        XCTAssertTrue(moveUntilFocused("episode.\(season).\(episode)", pressing: .down, limit: 6), "episode not focused; focus \(focusedIds())")
        press(.select, wait: 2)
    }
    private func open(_ film: String) {
        switch film {
        case "stargate": openStargate()
        case "ww": openPoster("poster.Wonder Woman", tab: "Movies"); pressPlay()
        case "divergent": openPoster("poster.Divergent", tab: "Movies"); pressPlay()
        default: openEpisode(show: "The Magicians", season: 1, episode: 1)
        }
    }
    private func attach() {
        app = XCUIApplication(); app.activate()
        note("attached at \(stamp())")
        XCTAssertTrue(app.buttons["tab.Movies"].waitForExistence(timeout: 60), "library did not load")
        sleep(3)
    }
    private func playerUp(_ tag: String) {
        XCTAssertTrue(app.otherElements["player"].waitForExistence(timeout: 20), "player did not appear; focus \(focusedIds())")
        note("\(tag): player up at \(stamp())")
    }

    private enum Ending: String { case click, playpause, menu }

    private func scrubSession(_ film: String) {
        let tag = "\(film)-scrub"
        attach(); open(film); playerUp(tag)
        sleep(30)
        for n in 1...4 { note("\(tag): skip +30 \(n) at \(stamp())"); remote.press(.right); usleep(1_500_000) }
        sleep(20)
        var isPaused = false
        let scrubs: [(Bool, Ending)] = [(true, .playpause), (false, .click), (true, .click), (false, .playpause), (true, .menu), (false, .menu)]
        for (i, (forward, ending)) in scrubs.enumerated() {
            let name = String(format: "p2d-\(tag)-%02d-\(forward ? "fwd" : "back")-\(ending.rawValue)", i + 1)
            if !isPaused { note("\(tag): pause (Play/Pause) before scrub \(i + 1) at \(stamp())"); press(.playPause, wait: 3); isPaused = true }
            note("\(tag): scrub \(i + 1) drag \(forward ? "forward" : "back") at \(stamp())")
            remote.press(forward ? .pageUp : .pageDown)
            usleep(1_000_000)
            shot("\(name)-a-mid-drag")
            sleep(4)
            note("\(tag): scrub \(i + 1) \(ending.rawValue) at \(stamp())")
            switch ending {
            case .click: remote.press(.select)
            case .playpause: remote.press(.playPause)
            case .menu: remote.press(.menu)
            }
            sleep(3)
            shot("\(name)-b-after+3s")
            if ending == .menu { sleep(7) } else { sleep(29); isPaused = false }
        }
        note("\(tag): exit at \(stamp())")
        press(.menu, wait: 3)
        XCTAssertFalse(app.otherElements["player"].exists, "player still up after the final Menu")
    }

    private func controlSession(_ film: String) {
        let tag = "\(film)-control"
        attach(); open(film); playerUp(tag)
        sleep(30)
        note("\(tag): playing drag forward at \(stamp())"); remote.press(.pageUp)
        usleep(1_000_000); shot("p2d-\(tag)-01-playing-drag-fwd-mid")
        sleep(4); shot("p2d-\(tag)-02-playing-drag-fwd-after")
        note("\(tag): playing drag back at \(stamp())"); remote.press(.pageDown)
        usleep(1_000_000); shot("p2d-\(tag)-03-playing-drag-back-mid")
        sleep(4); shot("p2d-\(tag)-04-playing-drag-back-after")
        note("\(tag): right (+30) at \(stamp())"); press(.right, wait: 3)
        note("\(tag): left (−10) at \(stamp())"); press(.left, wait: 3)
        note("\(tag): pause (Play/Pause) at \(stamp())"); press(.playPause, wait: 4)
        shot("p2d-\(tag)-fs-00-paused")
        for n in 1...10 {
            let left = n <= 5
            note("\(tag): click \(n) \(left ? "L" : "R") at \(stamp())")
            remote.press(left ? .left : .right)
            sleep(3)
            shot(String(format: "p2d-\(tag)-fs-%02d-\(left ? "L" : "R")", n))
            usleep(1_200_000)
        }
        note("\(tag): play at \(stamp())"); press(.playPause, wait: 5)
        press(.menu, wait: 3)
        XCTAssertFalse(app.otherElements["player"].exists, "player still up after Menu")
    }

    func t0_Probe() {
        let tag = "magicians-probe"
        attach(); open("magicians"); playerUp(tag)
        sleep(15)
        note("\(tag): pause at \(stamp())"); press(.playPause, wait: 3)
        note("\(tag): drag forward at \(stamp())"); remote.press(.pageUp)
        usleep(1_000_000); shot("p2d-\(tag)-01-mid-drag")
        sleep(4)
        note("\(tag): play/pause (land) at \(stamp())"); remote.press(.playPause)
        sleep(5); shot("p2d-\(tag)-02-after+5s")
        press(.menu, wait: 3)
    }

    func tStargateScrub() { scrubSession("stargate") }
    func tStargateControl() { controlSession("stargate") }
    func tWonderWomanScrub() { scrubSession("ww") }
    func tWonderWomanControl() { controlSession("ww") }
    func tDivergentScrub() { scrubSession("divergent") }
    func tDivergentControl() { controlSession("divergent") }
    func tMagiciansScrub() { scrubSession("magicians") }
    func tMagiciansControl() { controlSession("magicians") }
}
