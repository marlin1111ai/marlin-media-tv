### diag2e.py
# diag2e.py <run dir> [--excerpt N]: pass 2e, instrumented libvlc. Per paused scrub (pause → drag seeks → lift → land):
#   per seek: time since the pause at the seek, the laps it caused (2e: late_reset lines: late, total delay, now − pause
#   date), buffering ends and their anchor, demux PCR range (2e: pcr= lines, logged while paused), need-data traffic
#   (decoder notify by ES category, es_out forwarded/dropped, input flag transitions), pause-gate evaluations
#   (1h: check1 tuples), the pictures put while paused (2e: paused put / first picture pts);
#   then the landing: es_out change_pause, stop_buffering stream_start at Play, first pictures, VLCKit +1/+3 s times;
#   and from the trace: the first realtime video render after land (pts) and demux reads during the hold.
# Ticks: app tick= and VLC now= are CLOCK_MONOTONIC µs; tracer Timestamp is ns on the same clock.
# --excerpt N: print scrub N's log from the last seek to land + 20 lines, collapsing runs of identical-shape 1h/2e lines.
import json, re, sys, os, collections

d = sys.argv[1]
excerpt = int(sys.argv[sys.argv.index('--excerpt') + 1]) if '--excerpt' in sys.argv else None
log = open(os.path.join(d, 'marlin-media-tv.log'), errors='replace').read().split('\n')
st = [i for i, l in enumerate(log) if '[player] play() called' in l]; log = log[st[-1]:] if st else log

def num(pat, l, cast=int):
    m = re.search(pat, l); return cast(m.group(1)) if m else None

pauses = [i for i, l in enumerate(log) if re.search(r'\[player\] pause \(playPause button\)', l)]
begins = [i for i, l in enumerate(log) if '[scrub] begin at' in l]
segs = []
for b in begins:
    p = max([i for i in pauses if i < b], default=None)
    nxt = min([i for i in pauses if i > b], default=len(log))
    segs.append((p, b, nxt))

ev = []
tr = [f for f in os.listdir(d) if f.startswith('vlc-trace-')]
if tr:
    for line in open(os.path.join(d, tr[0]), errors='replace'):
        try: o = json.loads(line)
