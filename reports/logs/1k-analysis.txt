### matrix1k.py
# matrix1k.py <run dir>...: pass 1k step 2 numbers per run (clean recipe build, no instrumentation).
#   position at pause  = app "[player] state Paused at N ms" after the last "[player] pause (" line
#   position at Play   = app "[player] play (select) at N ms"
#   paused             = trace time from the video 'paused' event to the video 'resumed' event
#   dropped / late     = video RENDER 'toolate' / 'late' events in the 60 s after the video 'resumed' event
#   first picture      = the first realtime video RENDER after 'resumed': time (ms from resume), pts, drift
#   audio start        = NOT MEASURED (no timestamped audio-start event in the clean build; pass 1j §3, pass 1k §2).
#   traced instead     = the first audio DEC OUT after 'resumed' (timestamped decoder event): its time and pts, and the
#                        wall time that pts is scheduled for, through the first realtime video RENDER with |drift| < 5 ms:
#                        sched(pts_a) = t(render) − drift + (pts_a − pts_render); reported relative to the first picture.
#                        For a no-step run the audio renderer was not flushed, so blocks queued before the pause play
#                        first; the first DEC OUT is then a later block, and this figure is not the start of sound.
#   stepping           = count of "[framestep]" gotoPreviousFrame / gotoNextFrame lines
import json, re, sys, os

def ms(x): return None if x is None else round(x / 1e6, 1)

for d in sys.argv[1:]:
    tag = os.path.basename(d.rstrip('/'))
    log = open(os.path.join(d, 'marlin-media-tv.log'), errors='replace').read().split('\n')
    st = [i for i, l in enumerate(log) if '[player] play() called' in l]; log = log[st[-1]:] if st else log
    pz = [i for i, l in enumerate(log) if '[player] pause (' in l]
    pause_ms = next((re.search(r'state Paused at (\d+)', l).group(1) for l in log[pz[-1]:] if 'state Paused at' in l), None) if pz else None
    pl = [l for l in log if '[player] play (select) at' in l]
    play_ms = re.search(r'at (\d+) ms', pl[-1]).group(1) if pl else None
    back = sum(1 for l in log if '[framestep] −1 gotoPreviousFrame' in l)
    fwd = sum(1 for l in log if '[framestep] +1 gotoNextFrame' in l)
    spu = sorted(set(re.findall(r"ES track selected: '(spu/\d+)'", '\n'.join(log))))

    tr = [f for f in os.listdir(d) if f.startswith('vlc-trace-')][0]
    ev = []
    for line in open(os.path.join(d, tr), errors='replace'):
        try: o = json.loads(line)
        except Exception: continue
        b = o['Body']; b['_t'] = int(o['Timestamp']); ev.append(b)
    ev.sort(key=lambda b: b['_t'])
    vid = next((b['id'] for b in reversed(ev) if str(b.get('id', '')).startswith('video') and b.get('event') == 'resumed'), 'video/1')
    aid = next((b['id'] for b in ev if b.get('type') == 'DEC' and str(b.get('id', '')).startswith('audio')), None)
    ip = max(i for i, b in enumerate(ev) if b.get('id') == vid and b.get('event') == 'paused')
    ir = max(i for i, b in enumerate(ev) if b.get('id') == vid and b.get('event') == 'resumed')
    t0 = ev[ir]['_t']
    win = [b for b in ev[ir:] if b['_t'] <= t0 + 60e9]
    tl = sum(1 for b in win if b.get('id') == vid and b.get('event') == 'toolate')
    lt = sum(1 for b in win if b.get('id') == vid and b.get('event') == 'late')
    renders = [b for b in win if b.get('id') == vid and b.get('type') == 'RENDER' and b.get('mode') == 'realtime']
    fv = renders[0] if renders else None
    ontime = next((b for b in renders if abs(int(b.get('drift', 0))) < 5e6), None)
    fa = next((b for b in win if b.get('id') == aid and b.get('type') == 'DEC' and b.get('stream') == 'OUT'), None)
    sched = None
    if fa is not None and ontime is not None and fv is not None:
        t_sched = ontime['_t'] - int(ontime.get('drift', 0)) + (int(fa['pts']) - int(ontime['pts']))
        sched = ms(t_sched - fv['_t'])
    print(f'{tag}: spu {spu or "none"} | steps back {back} fwd {fwd} | pause at {pause_ms} ms, Play at {play_ms} ms, '
          f'paused {(t0 - ev[ip]["_t"]) / 1e9:.1f} s | 60 s after resume: dropped {tl}, late {lt} | '
          f'first picture +{ms(fv["_t"] - t0) if fv else None} ms pts {ms(int(fv["pts"])) if fv else None} drift {ms(int(fv.get("drift", 0))) if fv else None} | '
          f'first audio DEC OUT +{ms(fa["_t"] - t0) if fa else None} ms pts {ms(int(fa["pts"])) if fa else None}; '
          f'scheduled {sched} ms after the first picture (via on-time picture pts {ms(int(ontime["pts"])) if ontime else None})')

# exact1k.py <run dir>: pass 1k step 3 — one run of pass 1f's pattern (pause, 5 × L, 5 × R, 5 × L, 3.6 s apart).
# Per click, from the app log: key, before, callback time, after(400 ms). From the trace (video DEC OUT after the pause,
# grouped by gaps > 1 s): the picture shown per click by VLC's rules —
#   back step:    the decode run started by that click's seek; the picture displayed is the last DEC OUT whose PTS is
#                 below the previously displayed PTS (decoder_prevframe_AddPic);
#   forward step: the next picture in PTS order after the displayed one (already decoded, no new DEC OUT needed).
# The paused picture's PTS is taken as the first back step's pf_pts (the PTS its run ends at).
# Screenshots: mean absolute difference over the video area of every paused screenshot pair, 3840×2160 PNGs scaled to
# 960×540 by sips, crop x 130..830, y 120..380, excluding the pause glyph x 440..520, y 145..220 (pass 1f's picdiff.py).

st = [i for i, l in enumerate(log) if '[player] play() called' in l]; log = log[st[-1]:] if st else log
clicks = []
for l in log:
    m = re.search(r'\[framestep\] (−1 gotoPreviousFrame|\+1 gotoNextFrame) before=(\d+) ms', l)
    if m: clicks.append({'key': 'L' if m.group(1).startswith('−') else 'R', 'before': int(m.group(2))}); continue
    m = re.search(r'\[framestep\] (previousFrameStepped|nextFrameStepped) result=(-?\d+) time=(\d+) ms', l)
    if m and clicks: clicks[-1]['cb'] = f'{m.group(2)}/{m.group(3)}'; continue
    m = re.search(r'\[framestep\] [−+]1 after(?:\(400 ms\))?=(\d+) ms', l)
    if m and clicks: clicks[-1]['after'] = int(m.group(1))

# align app wall clock to trace time through the pause: app "[player] pause (" line  <->  trace vout 'paused' event
def appsec(l):
    m = re.match(r'^(\d+):(\d\d):(\d\d\.\d\d\d) ?\s*(AM|PM)', l)
    if not m: return None
    h = int(m.group(1)) % 12 + (12 if m.group(4) == 'PM' else 0)
    return h * 3600 + int(m.group(2)) * 60 + float(m.group(3))
pause_app = next(appsec(l) for l in log if '[player] pause (' in l)
click_t = [ev[ip]['_t'] + (appsec(l) - pause_app) * 1e9 for l in log
           if re.search(r'\[framestep\] (−1 gotoPreviousFrame|\+1 gotoNextFrame) before=', l)]
print(f'{tag}: {len(clicks)} clicks ({len(click_t)} timed); {len(after_pause)} video DEC OUT between pause and resume')
shown = None
rows = []
for n, c in enumerate(clicks, 1):

    pts = [int(b['pts']) // 1000000 for b in after_pause if lo <= b['_t'] < hi]   # ms
    if c['key'] == 'L':
        if shown is None and pts:
            shown = max(pts)                                       # the first back run decodes up to the paused picture
            rows.append((0, 'paused', None, None, None, shown, None, None))
        below = [p for p in pts if p < shown]
        new = max(below) if below else None
    else:
        later = [p for p in grid_ms if shown is not None and p > shown]
        new = min(later) if later else None
    step = None if new is None or shown is None else new - shown
    rows.append((n, c['key'], c['before'], c.get('cb'), c.get('after'), new, step, len(pts)))
    if new is not None: shown = new
for r in rows:

    print(f'  click {r[0]:2d} {r[1]}  before {r[2]}  callback {r[3]}  after(400 ms) {r[4]}  | trace: {r[7]} DEC OUT in window, picture pts {r[5]} ms, step {s}, grid positions moved {pos}')
steps = [r for r in rows if r[1] != 'paused']
