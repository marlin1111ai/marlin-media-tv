# gatecheck.py <app log>: last player session. Walks the log in order; tracks the last stamped app line.
# Prints every change of the check1 tuple (es_out_GetBuffering, b_eof, next_frame_need_data, demux_allowed) with its
# line, the nearest preceding app line, and how many evaluations the previous tuple held.
import re, sys, collections
lines = open(sys.argv[1], errors='replace').read().split('\n')
st = [i for i, l in enumerate(lines) if '[player] play() called' in l]
base = st[-1] if st else 0
last_app = ''; prev = None; held = 0; first_line = None
counts = collections.Counter()
for i in range(base, len(lines)):
    l = lines[i]
    if re.match(r'^\d+:\d\d:\d\d\.\d\d\d', l) and ('[player]' in l or '[framestep]' in l): last_app = l.strip()[:90]
    m = re.search(r'1h: check1 now=(\d+) state=PAUSE_S es_out_GetBuffering=(\d) b_eof=(\d) next_frame_need_data=(\d) -> demux_allowed=(\d)', l)
    if not m: continue
    tup = m.group(2, 3, 4, 5); counts[tup] += 1
    if tup != prev:
        if prev is not None: print(f'   … held {held} evaluations (log lines {first_line}..{i})')
        print(f'line {i+1}: GetBuffering={tup[0]} b_eof={tup[1]} need_data={tup[2]} -> demux_allowed={tup[3]}   [after: {last_app}]')
        prev = tup; held = 0; first_line = i + 1
    held += 1
print(f'   … held {held} evaluations (to end)')
print('totals (GetBuffering, b_eof, need_data, demux_allowed):', dict(counts))

# pacecheck.py <app log>: last session. From the first "[player] pause (" line: per 5 s of VLC tick time (the "now=" in 1h lines,
# µs, origin = the 1g es_out change_pause paused=1 date), the demuxed-while-paused count and wakeup-now (ms) min/median/max
# and how many had wakeup=0; plus every 1g stop_buffering anchor line (update, current_date) and its offset from the pause.
import re, sys, statistics as stt, collections
lines = open(sys.argv[1], errors='replace').read().split('\n')
st = [i for i, l in enumerate(lines) if '[player] play() called' in l]; lines = lines[st[-1]:] if st else lines
p0 = next(int(m.group(1)) for l in lines for m in [re.search(r'1g: es_out change_pause paused=1 date=(\d+)', l)] if m)
play = next((int(m.group(1)) for l in lines for m in [re.search(r'1g: input control_unpause date=(\d+)', l)] if m), None)
print(f'pause date (VLC ticks) {p0}; Play at +{(play - p0) / 1e6:.3f} s' if play else f'pause date {p0}')
B = collections.OrderedDict()
for l in lines:
    m = re.search(r'1h: demuxed-while-paused now=(\d+) wakeup=(-?\d+) wakeup-now=(-?\d+)', l)
    if m:
        now, wk, d = map(int, m.groups()); k = int((now - p0) // 5e6) * 5
        b = B.setdefault(k, {'n': 0, 'd': [], 'zero': 0})
        b['n'] += 1
        if wk == 0: b['zero'] += 1
        else: b['d'].append(d / 1000)
    m = re.search(r'1g: es_out stop_buffering now=(\d+) paused=(\d) pause_date=(-?\d+) current_date=(\d+) buffering_duration=(\d+) update=(\d+) stream_start=(\d+)', l)
    if m:
        now, pz, pd, cd, bd, up, ss = map(int, m.groups())
        print(f'  anchor at +{(now - p0) / 1e6:8.3f} s: paused={pz} current_date={"pause date" if cd == p0 else f"+{(cd - p0) / 1e6:.3f} s"} update=+{(up - p0) / 1e6:.3f} s stream_start={ss / 1000:.0f} ms')
for k, b in B.items():
    d = b['d']
    print(f'+{k:3d}s demux calls while paused {b["n"]:5d}  wakeup=0: {b["zero"]:5d}  wakeup-now ms: ' + (f'min {min(d):9.1f} median {stt.median(d):9.1f} max {max(d):9.1f}' if d else '-'))

### excerpt.py
# excerpt.py <full app log> <out>: every non-repetitive line; every check1 value transition; and, between two
# non-repetitive lines, only the first 12 and last 12 repetitive 1h lines (the rest is counted in a marker line).
import sys, re
lines = open(sys.argv[1], errors='replace').read().split('\n')
rep = re.compile(r'1h: (check1|check2|wait|demuxed-while-paused|decoder notify need_data=1|INPUT_CONTROL_NEED_DATA_FRAME_NEXT 1 -> 1)')
sig = re.compile(r'es_out_GetBuffering=\d b_eof=\d next_frame_need_data=\d -> demux_allowed=\d')
out = []; seg = []; last_tuple = None
def flush():
    if len(seg) <= 24:
        out.extend(l for _, l in seg)
    else:
        head, tail, mid = seg[:12], seg[-12:], seg[12:-12]
        out.extend(l for _, l in head)
        trans = [l for keep, l in mid if keep]
        out.append(f'[… {len(mid) - len(trans)} repetitive 1h lines omitted (full log: 1h-stargate-back-full.log.gz); check1 transitions kept below …]')
        out.extend(trans)
        out.extend(l for _, l in tail)
    seg.clear()
for l in lines:
    if rep.search(l):
        keep = False
        m = sig.search(l) if '1h: check1' in l else None
        if m and m.group(0) != last_tuple:
            keep = True; last_tuple = m.group(0)
        seg.append((keep, l))
    else:
        flush(); out.append(l)
flush()
open(sys.argv[2], 'w').write('\n'.join(out))
print('excerpt lines', len(out))
