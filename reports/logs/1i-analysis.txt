### matrix.py
# matrix.py <run dir>...: step 6 numbers per run (pass 1i). For each run folder written by runone.sh:
#   position at pause   = the app's "[player] state Paused at N ms" after the last "[player] pause (" line
#   position at Play    = the app's "[player] play (select) at N ms"
#   dropped / late      = video RENDER 'toolate' / 'late' trace events in the 60 s after the vout 'resumed' event
#   audio vs picture    = first realtime audio render − first realtime video render after 'resumed' (ms)
#   reads while paused  = DEMUX OUT events between vout 'paused' and 'resumed', split into those within 2.5 s after a
#                         step click ("[framestep]" app line) and the rest; highest PCR read while paused
#   anchor at Play      = VLC's own "Stream buffering done" after Play and the first too-late line
# Clock at Play+3 s / +60 s is read from the screenshots, not here.
import json, re, sys, os, datetime

def appsec(l):
    m = re.match(r'^(\d+):(\d\d):(\d\d\.\d\d\d) ?\s*(AM|PM)', l)
    if not m: return None
    h = int(m.group(1)) % 12 + (12 if m.group(4) == 'PM' else 0)
    return h * 3600 + int(m.group(2)) * 60 + float(m.group(3))

for d in sys.argv[1:]:
    tag = os.path.basename(d.rstrip('/'))
    log = open(os.path.join(d, 'marlin-media-tv.log'), errors='replace').read().split('\n')
    st = [i for i, l in enumerate(log) if '[player] play() called' in l]; log = log[st[-1]:] if st else log
    pz = [i for i, l in enumerate(log) if '[player] pause (' in l]
    pause_ms = next((re.search(r'state Paused at (\d+)', l).group(1) for l in log[pz[-1]:] if 'state Paused at' in l), None) if pz else None
    pl = [l for l in log if '[player] play (select) at' in l]
    play_ms = re.search(r'at (\d+) ms', pl[-1]).group(1) if pl else None
    steps = [appsec(l) for l in log if '[framestep]' in l and ('gotoPreviousFrame before' in l or 'gotoNextFrame before' in l)]
    pause_s = appsec(log[pz[-1]]) if pz else None; play_s = appsec(pl[-1]) if pl else None
    ip = next(i for i, l in enumerate(log) if '[player] play (select) at' in l) if pl else 0
    after = log[ip:ip + 3000]
    sbd = next((l.strip() for l in after if 'Stream buffering done' in l), None)
    firstlate = next((l.strip() for l in after if 'too late' in l or 'displayed late' in l), None)
    spu = sorted(set(re.findall(r"ES track selected: '(spu/\d+)'", '\n'.join(log))))

    ev = []
    tr = [f for f in os.listdir(d) if f.startswith('vlc-trace-')][0]
    for line in open(os.path.join(d, tr), errors='replace'):
        try: o = json.loads(line)
        except Exception: continue
        b = o['Body']; b['_t'] = int(o['Timestamp']); ev.append(b)
    ev.sort(key=lambda b: b['_t'])
    vid = next((b['id'] for b in reversed(ev) if str(b.get('id', '')).startswith('video') and b.get('event') == 'paused'), 'video/1')
    ip_ = max(i for i, b in enumerate(ev) if b.get('id') == vid and b.get('event') == 'paused')
    ir = max(i for i, b in enumerate(ev) if b.get('id') == vid and b.get('event') == 'resumed')
    t0 = ev[ip_]['_t']; tr0 = ev[ir]['_t']
    # map app wall clock to trace time through the pause/Play pair
    k = (tr0 - t0) / 1e9 / (play_s - pause_s) if play_s and pause_s and play_s > pause_s else 1
    stept = [t0 + (s - pause_s) * k * 1e9 for s in steps]
    near = far = 0; pcrmax = None; farpcr = []
    for b in ev[ip_:ir]:
        if b.get('type') == 'DEMUX' and b.get('id') == 'PCR':
            p = int(b['pcr']) // 1000000; pcrmax = p if pcrmax is None else max(pcrmax, p)
        if b.get('type') == 'DEMUX' and b.get('stream') == 'OUT':
            if any(0 <= b['_t'] - s <= 2.5e9 for s in stept): near += 1
            else: far += 1; farpcr.append((round((b['_t'] - t0) / 1e9, 1), b.get('id')))
    win = [b for b in ev[ir:] if b['_t'] <= tr0 + 60e9]
    aid = next((b['id'] for b in ev if b.get('type') == 'DEC' and str(b.get('id', '')).startswith('audio')), None)
    tl = sum(1 for b in win if b.get('id') == vid and b.get('event') == 'toolate')
    lt = sum(1 for b in win if b.get('id') == vid and b.get('event') == 'late')
    fv = next((b for b in win if b.get('id') == vid and b.get('type') == 'RENDER' and b.get('mode') == 'realtime'), None)
    fa = next((b for b in win if b.get('id') == aid and b.get('type') == 'RENDER' and b.get('mode') == 'realtime'), None)
    av = None if fv is None or fa is None else round((fa['_t'] - fv['_t']) / 1e6)
    print(f'{tag}: spu {spu or "none"} | pause at {pause_ms} ms, Play at {play_ms} ms, paused {(tr0 - t0) / 1e9:.1f} s | '
          f'steps {len(steps)} | reads while paused: {near} within 2.5 s of a step, {far} otherwise'
          f'{(" first " + str(farpcr[:3])) if far else ""}; highest PCR while paused {pcrmax} ms | '
          f'60 s after resume: dropped {tl}, late {lt}, first video drift {None if fv is None else round(int(fv.get("drift", 0)) / 1e6, 1)} ms, '
          f'audio − video first render {av} ms')
    print(f'    after Play: {sbd} | first late line: {firstlate}')
### pausereads.py
# pausereads.py <trace.json> [video id]: last session. Between the last vout 'paused' event and the next 'resumed':
# per 5 s bucket, the demux PCR range and DEMUX OUT counts per stream; then the totals, the highest PCR read while
# paused, the last PCR before the pause and the read-ahead (highest PCR while paused − last PCR before the pause).
# The input demuxes while paused only when es_out is buffering or next_frame_need_data is set (input.c:668–669), so a
# bucket with reads and no step in it marks the flag.
import json, sys, collections
vid = sys.argv[2] if len(sys.argv) > 2 else None
ev = []
for line in open(sys.argv[1], errors='replace'):
    try: o = json.loads(line)
    except Exception: continue
    b = o['Body']; b['_t'] = int(o['Timestamp']); ev.append(b)
ev.sort(key=lambda b: b['_t'])
if vid is None:
    vid = next((b['id'] for b in reversed(ev) if str(b.get('id', '')).startswith('video') and b.get('event') == 'paused'), 'video/1')
pz = max(i for i, b in enumerate(ev) if b.get('id') == vid and b.get('event') == 'paused')
rs = next((i for i in range(pz, len(ev)) if ev[i].get('id') == vid and ev[i].get('event') == 'resumed'), len(ev) - 1)
t0 = ev[pz]['_t']; tr = ev[rs]['_t']
pre = [int(b['pcr']) for b in ev[:pz] if b.get('type') == 'DEMUX' and b.get('id') == 'PCR']
print(f'{vid}: paused at trace t0; resumed at +{(tr - t0) / 1e9:.2f} s; last PCR before pause {pre[-1] // 1000 if pre else None} ms')
B = collections.OrderedDict(); tot = collections.Counter(); pcr_max = None
for b in ev[pz:rs]:
    k = int((b['_t'] - t0) // 5e9) * 5
    d = B.setdefault(k, {'pcr': [], 'out': collections.Counter()})
    if b.get('type') == 'DEMUX' and b.get('id') == 'PCR':
        p = int(b['pcr']); d['pcr'].append(p // 1000); pcr_max = p if pcr_max is None else max(pcr_max, p)
    if b.get('type') == 'DEMUX' and b.get('stream') == 'OUT':
        d['out'][str(b.get('id'))] += 1; tot[str(b.get('id'))] += 1
for k in range(0, int((tr - t0) / 1e9) + 5, 5):
    d = B.get(k, {'pcr': [], 'out': collections.Counter()})

#!/bin/sh
# runone.sh <test method> <run tag> — one traced attach-mode run on Home Theater (pass 1i).
# Console launch with VLCKit's tvOS defaults + JSON tracer into the app container, attach Diag1iUITests/<method>,
# export attachments, copy the app log and the trace off, stop the app.
S=<scratch>
REPO="$HOME/Xcode/Marlin Media TV"
T="$1"; TAG="$2"; C=$(cat "$S/container.txt")
case "$C" in /private/var/mobile/Containers/Data/Application/*) ;; *) echo "no container path — stop" >&2; exit 3;; esac
case "$T" in t[A-Z]*) ;; *) echo "bad test name '$T' — stop" >&2; exit 4;; esac
case "$TAG" in [a-z]*-[a-z]*) ;; *) echo "bad run tag '$TAG' — stop" >&2; exit 4;; esac
OUT="$S/runs/$TAG"; [ -e "$OUT" ] && { echo "$OUT exists — stop" >&2; exit 5; }; mkdir -p "$OUT"
P='("--no-color","--no-osd","--no-video-title-show","--no-snapshot-preview","--http-reconnect","--text-renderer=freetype","--avi-index=3","--audio-resampler=soxr","--tracer=json","--json-tracer-file='"$C"'/Library/Caches/vlc-trace-'"$TAG"'.json")'
echo "launch $(date '+%H:%M:%S')" | tee "$OUT/run.txt"
perl -e 'alarm 600; exec @ARGV' -- xcrun devicectl device process launch --console --terminate-existing --device "Home Theater" com.marlin1111.marlin-media-tv -VLCParams "$P" > "$OUT/console.out" 2>&1 &
LPID=$!
/bin/sleep 10
cd "$REPO"
xcodebuild test-without-building -project "Marlin Media TV.xcodeproj" -scheme "Marlin Media TV" \
  -destination 'platform=tvOS,name=Home Theater' -derivedDataPath build/DerivedData \
  -only-testing:"Marlin Media TVUITests/Diag1iUITests/$T" -resultBundlePath "$OUT/r.xcresult" > "$OUT/xcodebuild.out" 2>&1
echo "test exit $? $(date '+%H:%M:%S')" | tee -a "$OUT/run.txt"
grep -E "\[p1i\]|Test Case .*(passed|failed)|error:" "$OUT/xcodebuild.out" | sed -E 's/^.*\[p1i\] /[p1i] /' >> "$OUT/run.txt"
xcrun xcresulttool export attachments --path "$OUT/r.xcresult" --output-path "$OUT/att" > /dev/null 2>&1
for f in marlin-media-tv.log "vlc-trace-$TAG.json"; do
  xcrun devicectl device copy from --device "Home Theater" --domain-type appDataContainer \
    --domain-identifier com.marlin1111.marlin-media-tv --source "Library/Caches/$f" --destination "$OUT/$f" > /dev/null 2>&1 \
    || echo "copy failed: $f" | tee -a "$OUT/run.txt"
done
kill $LPID 2>/dev/null; wait $LPID 2>/dev/null
ls -la "$OUT" | awk '{print $5, $9}' | tee -a "$OUT/run.txt"
grep -E "Test Case .*(passed|failed)" "$OUT/xcodebuild.out"
