### resume1j.py
# resume1j.py <app log>: last player session of a pass 1j instrumented run (1g + 1h + 1j lines).
# For every "[player] play (select)" line: the resume sequence in log order up to 2.5 s of VLC ticks after the unpause —
# stepping stop (pf_pts, fifo), es_out change_pause/position, input clock reference, stop_buffering / set_first_pcr,
# the first 12 pictures put to the vout, every vout drop / late render with its pts, the audio decoder's first blocks,
# the audio core's first play dates and timing points, the tvOS audio output's first play / startNow / first
# time_observed. Times are printed relative to the 1g "input control_unpause" date (ms).
# Then, per pause segment, the need-data flag writes and the check1 gate totals (as flagcheck.py).
import re, sys, collections

lines = open(sys.argv[1], errors='replace').read().split('\n')
st = [i for i, l in enumerate(lines) if '[player] play() called' in l]
lines = lines[st[-1]:] if st else lines
keep = re.compile(r'1j: |1g: (es_out change_pause|es_out change_position|es_out stop_buffering|set_first_pcr|main_change_pause|main_reset|input control_unpause)|Stream buffering done|deferring start|starting late|\] started$|\[player\] (play|pause) \(|picture is too late|picture displayed late|discontinuity')
ticks = re.compile(r'now=(\d+)')
plays = [i for i, l in enumerate(lines) if '[player] play (select)' in l]
for k, ip in enumerate(plays):
    # find the unpause date after this play line
    t0 = None
    for l in lines[ip:ip + 200]:
        m = re.search(r'1g: input control_unpause date=(\d+)', l)
        if m: t0 = int(m.group(1)); break
    print(f'\n=== Play {k + 1}: {lines[ip].strip()}  (unpause date {t0})')
    end = plays[k + 1] if k + 1 < len(plays) else len(lines)
    video_put = 0; shown = 0
    for l in lines[ip:end]:
        if not keep.search(l): continue
        m = ticks.search(l); rel = None
        if m and t0: rel = (int(m.group(1)) - t0) / 1000
        if rel is not None and rel > 2500: break
        if '1j: video put' in l:
            video_put += 1
            if video_put > 12: continue
        tag = f'{rel:+9.1f} ms ' if rel is not None else '             '
        print(tag + l.strip()[:230]); shown += 1
    print(f'    ({video_put} "video put" lines in the window, 12 shown)')

print('\n=== need-data flag and gate per pause segment')
seg = None; segs = []
for l in lines:
    if '[player] pause (select)' in l:
        seg = {'start': l.strip()[:80], 'writes': [], 'check1': collections.Counter(), 'reads': 0}; segs.append(seg)
    elif '[player] play (select)' in l:
        if seg: seg['end'] = l.strip()[:80]
        seg = None
    if seg is None: continue
    m = re.search(r'1h: INPUT_CONTROL_NEED_DATA_FRAME_NEXT (\d) -> (\d)', l)
    if m: seg['writes'].append(f'{m.group(1)}->{m.group(2)}')
    if '1h: next_frame_need_data cleared' in l: seg['writes'].append('cleared')
    m = re.search(r'1h: check1 .* es_out_GetBuffering=(\d) b_eof=(\d) next_frame_need_data=(\d) -> demux_allowed=(\d)', l)
    if m: seg['check1'][m.group(1, 2, 3, 4)] += 1
    if '1h: demuxed-while-paused' in l: seg['reads'] += 1
for s in segs:
    w = collections.Counter(s['writes'])
    print(f'[{s["start"]}] -> [{s.get("end", "?")}]: flag writes {dict(w)}; check1 (GetBuffering,b_eof,need_data,demux_allowed) {dict(s["check1"])}; demuxed-while-paused {s["reads"]}')
