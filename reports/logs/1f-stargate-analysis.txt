### stepcheck.py
# stepcheck.py <trace.json>: last session in the trace (after the last 'start wait' gap reset); video/1.
# Displayed picture per previous-frame step = VLC's rule (decoder_prevframe_AddPic): the stored picture shown when a
# decoded picture with pts >= pf_pts arrives; in the trace that is the DEC OUT just before the step's last DEC OUT.
# Next-frame step: the one new DEC OUT, or the resume picture kept from the previous step.
import json, sys
ev = []
for line in open(sys.argv[1], errors='replace'):
    try: o = json.loads(line)
    except Exception: continue
    b = o['Body']
    vid = sys.argv[2] if len(sys.argv) > 2 else 'video/1'
    if str(b.get('id', '')) != vid: continue
    b['_t'] = int(o['Timestamp']); ev.append(b)
ev.sort(key=lambda b: b['_t'])
pz = max(i for i, b in enumerate(ev) if b.get('event') == 'paused')
gaps = [i for i in range(1, pz) if ev[i]['_t'] - ev[i-1]['_t'] > 600e9]
if gaps: ev = ev[gaps[-1]:]; pz -= gaps[-1]
t0 = ev[pz]['_t']
play = sorted({int(b['pts']) // 1000 for b in ev[:pz] if b.get('type') == 'DEC' and b.get('stream') == 'OUT'})
print('decoded PTS grid before pause (last 16, ms):', play[-16:])
print('  spacings:', [b - a for a, b in zip(play[-16:], play[-15:])])
# group post-pause activity into bursts separated by > 1 s
bursts, cur = [], []
for b in ev[pz + 1:]:
    if cur and b['_t'] - cur[-1]['_t'] > 1e9: bursts.append(cur); cur = []
    cur.append(b)
if cur: bursts.append(cur)
for k, bu in enumerate(bursts):
    outs = [int(b['pts']) // 1000 for b in bu if b.get('type') == 'DEC' and b.get('stream') == 'OUT']
    evs = [b['event'] for b in bu if b.get('event')]
    rend = sum(1 for b in bu if b.get('type') == 'RENDER' and b.get('mode') == 'realtime')
    print(f'burst {k+1:2d} at +{(bu[0]["_t"]-t0)/1e9:7.3f} s ({(bu[-1]["_t"]-bu[0]["_t"])/1e6:6.1f} ms): OUT {outs[:40]}{" …" if len(outs) > 40 else ""} (n={len(outs)}) events {evs[:12]}{" …" if len(evs)>12 else ""} realtime-renders {rend}')
### stepcheck2.py
# stepcheck2.py <trace.json> [video id]: last session; after the last 'paused' event, DEC OUT grouped by gaps > 1 s in
# DEC OUT time. Per group: all PTS, flush events inside, and VLC's prev-frame rule (the picture displayed is the stored
# one released when a PTS >= pf_pts arrives = the second-to-last OUT of a back step). Then the first realtime RENDER
# drifts after the resume (first DEC OUT group with > 20 pictures).
import json, sys
vid = sys.argv[2] if len(sys.argv) > 2 else 'video/1'
ev = []
for line in open(sys.argv[1], errors='replace'):
    try: o = json.loads(line)
    except Exception: continue
    b = o['Body']
    if str(b.get('id', '')) != vid: continue
    b['_t'] = int(o['Timestamp']); ev.append(b)
ev.sort(key=lambda b: b['_t'])
pz = max(i for i, b in enumerate(ev) if b.get('event') == 'paused')
t0 = ev[pz]['_t']
post = ev[pz + 1:]
groups, cur, last_out = [], [], None
for b in post:
    is_out = b.get('type') == 'DEC' and b.get('stream') == 'OUT'
    if is_out and last_out is not None and b['_t'] - last_out > 1e9 and cur: groups.append(cur); cur = []
    if is_out: last_out = b['_t']
    if is_out or b.get('event') in ('flushed', 'late', 'toolate'): cur.append(b)
if cur: groups.append(cur)
for k, g in enumerate(groups):
    seq = [('F' if b.get('event') == 'flushed' else b['event']) if b.get('event') else int(b['pts']) // 1000 for b in g]
    outs = [s for s in seq if isinstance(s, int)]
    print(f'group {k+1:2d} +{(g[0]["_t"]-t0)/1e9:7.3f} s: n={len(outs)}', seq if len(seq) <= 30 else seq[:30] + ['…'])
    if len(outs) > 20:
        rs = [(b['_t'], int(b['drift'])) for b in post if b.get('type') == 'RENDER' and b.get('mode') == 'realtime' and b['_t'] >= g[0]['_t']][:8]
        print('   first realtime RENDER drifts after resume (ms):', [round(d / 1e6, 1) for _, d in rs])
        print('   toolate events in group:', sum(1 for b in g if b.get('event') == 'toolate'), 'first/last OUT pts ms', outs[0], outs[-1], 'over', round((g[-1]['_t'] - g[0]['_t']) / 1e9, 2), 's')
        break
### picdiff.py
# picdiff.py <dir with 960-wide PNGs> <tag>: converts to 24-bit BMP with sips, then mean absolute difference (0-255) over the
# video area above the overlay (x 130..830, y 120..380 of 960×540), excluding the pause glyph (x 440..520, y 145..220).
import os, subprocess, struct, sys, itertools
d, tag = sys.argv[1], sys.argv[2]
names = sorted(n for n in os.listdir(d) if n.endswith('.png') and '-resumed' not in n)
px = {}
for n in names:
    bmp = os.path.join(d, n[:-4] + '.bmp')
    subprocess.run(['sips', '-s', 'format', 'bmp', os.path.join(d, n), '--out', bmp], capture_output=True)
    data = open(bmp, 'rb').read()
    off, = struct.unpack_from('<I', data, 10); w, h = struct.unpack_from('<ii', data, 18); bpp, = struct.unpack_from('<H', data, 28)
    step = bpp // 8; row = (w * step + 3) & ~3
    def at(x, y, data=data, off=off, h=h, row=row, step=step):
        yy = (h - 1 - y) if h > 0 else y
        i = off + yy * row + x * step
        return data[i], data[i + 1], data[i + 2]
    px[n] = [sum(at(x, y)) for y in range(120, 380, 2) for x in range(130, 830, 2) if not (440 <= x < 520 and 145 <= y < 220)]
short = [n.replace(f'p1f-{tag}-', '').replace('.png', '') for n in names]
print('MAD ×10 (0 = identical picture area)')
print('      ' + ' '.join(f'{s:>5}' for s in short))
for a, sa in zip(names, short):
    print(f'{sa:>5} ' + ' '.join(f'{round(10 * sum(abs(p - q) for p, q in zip(px[a], px[b])) / (3 * len(px[a]))):5d}' for b in names))
