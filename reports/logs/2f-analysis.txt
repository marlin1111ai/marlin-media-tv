### scrub2f.py
# scrub2f.py <run dir>... [MD=1 env for table rows]: pass 2f step 6, one line per paused scrub (0021 build, clean framework).
#   app log (from the last "[player] play() called"), per scrub, in order:
#     [player] pause (playPause button) at P ms      the pause before the scrub (its tick is not logged; the trace's vout
#                                                    'paused' event before begin gives the time paused before the drag)
#     [scrub] begin at S ms tick=T                   the drag's start
#     [scrub] seek X ms tick=T                       each paused seek during the drag and on lift
#     [scrub] lift target=X ms tick=T                the target shown when the thumb lifts
#     [scrub] land (select|playPause) from S ms at X ms tick=T  /  [scrub] cancel menu at X ms, back to S ms tick=T
#     [scrub] land|cancel +3 s time=N ms             VLCKit's time 3 s after the ending
#   tick = CLOCK_MONOTONIC µs = VLC's vlc_tick_now(); tracer Timestamp = ns on the same clock.
#   tracked            = for each paused seek of the drag (not the lift seek): a video RENDER (any mode) with pts within
#                        1 s of that seek's target before the next seek; "k/n". And the hold: every video RENDER from the
#                        lift seek + its first picture to the ending carries a pts within 1 s of the target ("hold at
#                        target yes/no", with the pts range).
#   press → picture    = first realtime video RENDER after the ending whose pts is within 2 s of the target (cancel: of
#                        the start); its time − ending tick. first picture − target (ms).
#   +3 s               = pts of the last realtime render at or before ending + 3 s (trace), and VLCKit's +3 s time.
#   dropped / late     = video RENDER 'toolate' / 'late' from that picture for 30 s (or up to the next pause line's scrub).
#   late resets        = VLC log "ES_OUT_SET_(GROUP_)PCR  is called … late" lines between begin and the ending (log order).
#   reads while paused = DEMUX OUT events from begin to the ending; demux PCR range; the last PCR − target (ms).
import json, re, sys, os

for d in sys.argv[1:]:
    tag = os.path.basename(d.rstrip('/'))
    log = open(os.path.join(d, 'marlin-media-tv.log'), errors='replace').read().split('\n')

        except Exception: continue
        b = o['Body']; b['_t'] = int(o['Timestamp']); ev.append(b)
    ev.sort(key=lambda x: x['_t'])
    vid = 'video/1'
    # trace pts and pcr are ns; kept here as µs
    renders = [(x['_t'], int(x['pts']) // 1000, x.get('mode')) for x in ev if x.get('id') == vid and x.get('type') == 'RENDER' and 'pts' in x]

    scrubs, cur = [], None
    for i, l in enumerate(log):
        m = re.search(r'\[scrub\] begin at (\d+) ms tick=(\d+)', l)
        if m: cur = {'start': int(m.group(1)), 'tb': int(m.group(2)) * 1000, 'ib': i, 'seeks': []}; scrubs.append(cur); continue
        if cur is None: continue

        if 'te' not in s: print(f'  {n}: no ending'); continue
        nxt = scrubs[n]['tb'] if n < len(scrubs) else ev[-1]['_t']
        land = s['kind'].startswith('land')

        lift_to_pic = (first_at_target - ls_t) / 1e6 if first_at_target else None
        # landing / cancel

        after = [(rt, p) for rt, p, mo in renders if s['te'] <= rt < nxt and mo == 'realtime']
        pic = next(((rt, p) for rt, p in after if abs(p - aim * 1000) <= 2_000_000), None)
        c2p = off = drops = lates = pts3 = None
        if pic:
            c2p = round((pic[0] - s['te']) / 1e9, 3); off = round((pic[1] - aim * 1000) / 1000)
            win = [x for x in ev if pic[0] <= x['_t'] <= min(pic[0] + 30e9, nxt) and x.get('id') == vid]
            drops = sum(1 for x in win if x.get('event') == 'toolate'); lates = sum(1 for x in win if x.get('event') == 'late')
        at3 = [(rt, p) for rt, p in after if rt <= s['te'] + 3e9]
        if at3: pts3 = at3[-1][1] // 1000   # ms
        err3 = (pts3 - (aim + max(0, 3000 - int(c2p * 1000)))) if (land and pts3 is not None and c2p is not None) else ((pts3 - aim) if pts3 is not None else None)
        if not land and s.get('t3') is not None: err3 = s['t3'] - aim
        late_lines = sum(1 for l in log[s['ib']:s['ie']] if 'is called' in l and 'late' in l)
        resets = sum(1 for l in log[s['ib']:s['ie']] if 'ES_OUT_RESET_PCR called' in l)
        reads = sum(1 for t in outs if s['tb'] <= t <= s['te'])
        wp = [v for t, v in pcrs if s['tb'] <= t <= s['te']]
        ahead = (wp[-1] // 1000 - target) if wp else None
        ending = {'land select': 'click', 'land playPause': 'Play/Pause', 'cancel menu': 'Menu'}[s['kind']]
        direction = 'fwd' if target > s['start'] else 'back'
        if os.environ.get('MD'):
            print(f"| {tag} | {n} | {paused_before:.0f} s | {direction} | {ending} | {s['start']} | {target} | {hits}/{len(drag)}, lift → picture {lift_to_pic} ms, hold {'at target' if hold_ok else 'NOT at target'} {hold_pts} | "
                  f"{pts3} ({err3}) / {s.get('t3')} | {c2p} | {off} | {drops} / {lates} | {late_lines} | {reads} (last PCR − target {ahead} ms) |")

# fs2f.py <run dir>: pass 2f frame-step check (pass 1k's picdiff method). Screenshots p2d-<tag>-fs-00-paused,
# fs-01..05-L, fs-06..10-R from the xcresult manifest, scaled to 960×540 BMP with sips; mean absolute difference ×10

# One picture per click: each click differs from the one before; the right clicks retrace the left ones —
# R6 = L4, R7 = L3, R8 = L2, R9 = L1, R10 = paused (MAD 0). Also the app's "[framestep] … after(400 ms)" positions.
import json, os, re, struct, subprocess, sys
d = sys.argv[1]
tmp = os.path.join(d, 'fsbmp'); os.makedirs(tmp, exist_ok=True)
shots = {}
for t in json.load(open(os.path.join(d, 'att', 'manifest.json'))):
    for a in t.get('attachments', []):
        n = a.get('suggestedHumanReadableName', '')
        m = re.search(r'-fs-(\d\d)-(\w+)_0_', n)
        if m: shots[int(m.group(1))] = (m.group(2), os.path.join(d, 'att', a['exportedFileName']))
px = {}
for k, (label, src) in sorted(shots.items()):
    bmp = os.path.join(tmp, f'{k:02d}.bmp')
    subprocess.run(['sips', '-s', 'format', 'bmp', '-z', '540', '960', src, '--out', bmp], capture_output=True)
    data = open(bmp, 'rb').read()
    off, = struct.unpack_from('<I', data, 10); w, h = struct.unpack_from('<ii', data, 18); bpp, = struct.unpack_from('<H', data, 28)
    step = bpp // 8; row = (w * step + 3) & ~3
    def at(x, y):
        yy = (h - 1 - y) if h > 0 else y
        i = off + yy * row + x * step
        return data[i] + data[i + 1] + data[i + 2]
    px[k] = [at(x, y) for y in range(120, 380, 2) for x in range(130, 830, 2) if not (440 <= x < 520 and 145 <= y < 220)]
def mad(a, b): return round(10 * sum(abs(p - q) for p, q in zip(px[a], px[b])) / (3 * len(px[a])))
names = {k: (f'{k:02d}-{v[0]}') for k, v in shots.items()}
print('MAD ×10 (0 = identical picture area)')
print('         ' + ' '.join(f'{names[k][:6]:>6}' for k in sorted(px)))
for a in sorted(px):
    print(f'{names[a]:>8} ' + ' '.join(f'{mad(a, b):6d}' for b in sorted(px)))
steps_differ = all(mad(k, k - 1) > 0 for k in range(1, 11) if k in px and k - 1 in px)
retrace = {6: 4, 7: 3, 8: 2, 9: 1, 10: 0}
retrace_ok = all(mad(r, l) == 0 for r, l in retrace.items() if r in px and l in px)
print(f'each click differs from the previous picture: {steps_differ} ({[mad(k, k - 1) for k in range(1, 11) if k in px and k - 1 in px]})')
print(f'right clicks retrace the left clicks (R6=L4 … R10=paused, MAD 0): {retrace_ok} ({[(r, l, mad(r, l)) for r, l in retrace.items() if r in px and l in px]})')
log = open(os.path.join(d, 'marlin-media-tv.log'), errors='replace').read()
print('app positions:', re.findall(r'\[framestep\] ([−+]1) after\(400 ms\)=(\d+) ms', log))

import json, re, sys, os

for d in sys.argv[1:]:
    tag = os.path.basename(d.rstrip('/'))

    if len(req) < 2: print('  second player not found'); continue
    a, b = req[-2], req[-1]
    first, second = log[a:b], log[b:]
    t = lambda l: re.sub(r'^.*?\[player\] ', '', l)[:150]
    for l in first:
        if re.search(r'\[player\] (request|pause \(|dismiss|state (Stopped|Error))', l): print('  1st | ' + l[:170])
    for l in second[:4000]:
        if re.search(r'\[player\] (request|play\(\) called|state (Opening|Buffering|Playing|Error))|Playback failed', l): print('  2nd | ' + l[:170])
        if '[player] state Playing' in l: break
    late = sum(1 for l in second if 'is called' in l and 'late' in l)
    resets = sum(1 for l in second if 'ES_OUT_RESET_PCR called' in l)
    errs = [l for l in second if 'Playback failed' in l or 'state Error' in l]
    ev = []
    tr = [f for f in os.listdir(d) if f.startswith('vlc-trace-')]
    for line in open(os.path.join(d, tr[0]), errors='replace'):
        try: o = json.loads(line)
        except Exception: continue
        x = o['Body']; x['_t'] = int(o['Timestamp']); ev.append(x)
    ev.sort(key=lambda x: x['_t'])
    rend = [x for x in ev if str(x.get('id', '')).startswith('video') and x.get('type') == 'RENDER' and x.get('mode') == 'realtime' and 'pts' in x]
    # the second title's renders: after the largest drop in pts (the first title was ~30 s in, the second starts near 0)
    start = next((k for k in range(1, len(rend)) if int(rend[k]['pts']) < int(rend[k - 1]['pts']) - 10_000_000), None)
    if start is None: print('  trace: no restart of pts found'); continue
    r0 = rend[start]
    vid = r0['id']
    win = [x for x in ev if r0['_t'] <= x['_t'] <= r0['_t'] + 30e9 and x.get('id') == vid]
    drops = sum(1 for x in win if x.get('event') == 'toolate'); lates = sum(1 for x in win if x.get('event') == 'late')
    at30 = [x for x in rend[start:] if x['_t'] <= r0['_t'] + 30e9]
    print(f"  second title: first picture pts {int(r0['pts']) / 1e6:.3f} ms; last pts at its +30 s {int(at30[-1]['pts']) / 1e6:.0f} ms; "  # trace pts is ns
          f"dropped {drops}, late {lates} in 30 s | PCR-late lines {late}, RESET_PCR {resets} | errors {len(errs)}")
### shots2f.py
# shots2f.py <run dir>: screenshots named from the xcresult manifest -> png/ (symlinks, full size) and jpg/ (960 wide).
import json, os, subprocess, sys
R = sys.argv[1]
os.makedirs(f'{R}/png', exist_ok=True); os.makedirs(f'{R}/jpg', exist_ok=True)
for t in json.load(open(f'{R}/att/manifest.json')):
    for a in t.get('attachments', []):
        n = a.get('suggestedHumanReadableName', '')
        if not n.startswith('p2f-'): continue
        short = n.split('_0_')[0]; src = os.path.abspath(f"{R}/att/{a['exportedFileName']}")
        if not os.path.exists(f'{R}/png/{short}.png'): os.symlink(src, f'{R}/png/{short}.png')
        subprocess.run(['sips', '-s', 'format', 'jpeg', '-Z', '960', src, '--out', f'{R}/jpg/{short}.jpg'], capture_output=True)
### runone2f.sh
#!/bin/sh
# runone2f.sh <test method> <run tag> — one traced attach-mode run on Home Theater (pass 2f; from pass 2e).
# Console launch with VLCKit's tvOS defaults + JSON tracer into the app container, attach Diag2fUITests/<method>,
# export attachments, copy the app log and the trace off, stop the app. Refuses empty or malformed names and never
# deletes or reuses a run folder.
S=<scratch>/2f
REPO="$HOME/Xcode/Marlin Media TV"
T="$1"; TAG="$2"; C=$(cat "$S/container.txt" 2>/dev/null)
case "$C" in /private/var/mobile/Containers/Data/Application/*) ;; *) echo "no container path — stop" >&2; exit 3;; esac
case "$T" in t[A-Z0-9]*) ;; *) echo "bad test name '$T' — stop" >&2; exit 4;; esac
case "$TAG" in [a-z]*-[a-z]*) ;; *) echo "bad run tag '$TAG' — stop" >&2; exit 4;; esac
OUT="$S/runs/$TAG"; [ -e "$OUT" ] && { echo "$OUT exists — stop" >&2; exit 5; }; mkdir -p "$OUT"
P='("--no-color","--no-osd","--no-video-title-show","--no-snapshot-preview","--http-reconnect","--text-renderer=freetype","--avi-index=3","--audio-resampler=soxr","--tracer=json","--json-tracer-file='"$C"'/Library/Caches/vlc-trace-'"$TAG"'.json")'
echo "launch $(date '+%H:%M:%S')" | tee "$OUT/run.txt"
perl -e 'alarm 1200; exec @ARGV' -- xcrun devicectl device process launch --console --terminate-existing --device "Home Theater" com.marlin1111.marlin-media-tv -VLCParams "$P" > "$OUT/console.out" 2>&1 &
LPID=$!
/bin/sleep 10
cd "$REPO"
xcodebuild test-without-building -project "Marlin Media TV.xcodeproj" -scheme "Marlin Media TV" \
  -destination 'platform=tvOS,name=Home Theater' -derivedDataPath build/DerivedData \
  -only-testing:"Marlin Media TVUITests/Diag2fUITests/$T" -resultBundlePath "$OUT/r.xcresult" > "$OUT/xcodebuild.out" 2>&1
echo "test exit $? $(date '+%H:%M:%S')" | tee -a "$OUT/run.txt"
grep -E "\[p2f\]|Test Case .*(passed|failed)|error:" "$OUT/xcodebuild.out" | sed -E 's/^.*\[p2f\] /[p2f] /' >> "$OUT/run.txt"
xcrun xcresulttool export attachments --path "$OUT/r.xcresult" --output-path "$OUT/att" > /dev/null 2>&1
for f in marlin-media-tv.log "vlc-trace-$TAG.json"; do
  xcrun devicectl device copy from --device "Home Theater" --domain-type appDataContainer \
    --domain-identifier com.marlin1111.marlin-media-tv --source "Library/Caches/$f" --destination "$OUT/$f" > /dev/null 2>&1 \
    || echo "copy failed: $f" | tee -a "$OUT/run.txt"
done
kill $LPID 2>/dev/null; wait $LPID 2>/dev/null
ls -la "$OUT" | awk '{print $5, $9}' | tee -a "$OUT/run.txt"
grep -E "Test Case .*(passed|failed)" "$OUT/xcodebuild.out"
