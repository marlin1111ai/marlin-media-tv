### scrub2a.py
# scrub2a.py <run dir>...: pass 2a numbers per scrub run (clean recipe build, no instrumentation).
#   app log (from the last "[player] play() called"): [scrub] begin / seek / lift / land|cancel lines with tick=µs on
#   CLOCK_MONOTONIC (VLC's vlc_tick_now); "+1 s" / "+3 s" time lines; [skip] lines.
#   trace: JSON tracer events, Timestamp in ns on the same clock (tick × 1000).
#   position before the drag = begin "at N ms"; target shown = land/cancel target; after landing = "+3 s time=".
#   clock check              = for a drag while playing, the video 'paused' event after begin, minus begin's tick.
#   demux reads while paused = DEMUX OUT events (every ES) from begin to land/cancel, split into
#                              drag (begin → lift + 2 s: the seeks' own reads) and hold (lift + 2 s → land: nothing
#                              should read), with the demux PCR range in each and the hold's PCR end minus the target.
#   ES_OUT_RESET_PCR         = VLC log lines between the begin and land/cancel lines (log order; VLC lines carry no time).
#   click → picture          = first realtime video RENDER after the video 'resumed' event that follows land, minus land's tick.
#   dropped / late           = video RENDER 'toolate' / 'late' in the 30 s after that 'resumed' event.
import json, re, sys, os

def sec(ns): return None if ns is None else round(ns / 1e9, 3)

for d in sys.argv[1:]:
    tag = os.path.basename(d.rstrip('/'))
    log = open(os.path.join(d, 'marlin-media-tv.log'), errors='replace').read().split('\n')
    st = [i for i, l in enumerate(log) if '[player] play() called' in l]; log = log[st[-1]:] if st else log
    def first(pat, lines=log):
        for i, l in enumerate(lines):
            m = re.search(pat, l)
            if m: return i, m
        return None, None
    ib, b = first(r'\[scrub\] begin at (\d+) ms playing=(\w+) tick=(\d+)')
    if b is None:
        print(f'{tag}: no [scrub] begin line'); continue
    il, lift = first(r'\[scrub\] lift target=(\d+) ms tick=(\d+)')
    ie, end = first(r'\[scrub\] (land) from (\d+) ms at (\d+) ms tick=(\d+)')
    kind = 'land'
    if end is None:
        ie, end = first(r'\[scrub\] (cancel) at (\d+) ms, back to (\d+) ms playing=(\w+) tick=(\d+)')
        kind = 'cancel'
    seeks = [re.search(r'seek (\d+) ms', l).group(1) for l in log if '[scrub] seek ' in l]
    after = {m.group(1): m.group(2) for m in (re.search(r'\[scrub\] (?:land|cancel) \+(\d) s time=(\d+) ms', l) for l in log) if m}
    ignored = [l.split('[scrub] ')[1] for l in log if 'ignored while scrubbing' in l or 'drag not taken' in l]
    resets = sum(1 for l in log[ib:ie or len(log)] if 'ES_OUT_RESET_PCR' in l)
    start_ms, playing, t_begin = int(b.group(1)), b.group(2) == 'true', int(b.group(3)) * 1000
    t_lift = int(lift.group(2)) * 1000 if lift else None
    if end is None:
        print(f'{tag}: begin at {start_ms} ms, no land/cancel line'); continue
    if kind == 'land': target, t_end = int(end.group(3)), int(end.group(4)) * 1000
    else: target, t_end = int(end.group(2)), int(end.group(5)) * 1000

    tr = [f for f in os.listdir(d) if f.startswith('vlc-trace-')]
    ev = []
    for line in open(os.path.join(d, tr[0]), errors='replace'):
        try: o = json.loads(line)
        except Exception: continue
        bb = o['Body']; bb['_t'] = int(o['Timestamp']); ev.append(bb)
    ev.sort(key=lambda x: x['_t'])
    vid = 'video/1'
    pz = next((x for x in ev if x['_t'] >= t_begin - 200e6 and x.get('id') == vid and x.get('event') == 'paused'), None)
    clock = f"paused event {sec(pz['_t'] - t_begin)} s after begin" if (pz and playing) else 'n/a (already paused)'
    split = (t_lift + 2e9) if t_lift else t_end
    def reads(a, z):
        w = [x for x in ev if a <= x['_t'] <= z]
        out = sum(1 for x in w if x.get('type') == 'DEMUX' and x.get('stream') == 'OUT')
        pcr = [int(x['pcr']) // 1000 for x in w if x.get('type') == 'DEMUX' and x.get('id') == 'PCR']
        return out, (pcr[0], pcr[-1]) if pcr else None
    dr, dpcr = reads(t_begin, split)
    hr, hpcr = reads(split, t_end)
    ahead = (hpcr[1] - target) if hpcr else None
    res = next((x for x in ev if x['_t'] >= t_end and x.get('id') == vid and x.get('event') == 'resumed'), None)
    pic = drops = lates = None
    if res:
        pic_ev = next((x for x in ev if x['_t'] >= res['_t'] and x.get('id') == vid and x.get('type') == 'RENDER' and x.get('mode') == 'realtime'), None)
        pic = sec(pic_ev['_t'] - t_end) if pic_ev else None
        win = [x for x in ev if res['_t'] <= x['_t'] <= res['_t'] + 30e9 and x.get('id') == vid]
        drops = sum(1 for x in win if x.get('event') == 'toolate'); lates = sum(1 for x in win if x.get('event') == 'late')
    print(f'{tag}: {kind} | playing at drag {playing} | before {start_ms} ms | seeks {len(seeks)} last {seeks[-1] if seeks else None} | '
          f'target {target} ms | after +1 s {after.get("1")} +3 s {after.get("3")} ms | {clock} | '
          f'reads drag {dr} PCR {dpcr} | reads hold {hr} PCR {hpcr} (hold PCR end − target {ahead} ms) | RESET_PCR lines {resets} | '
          f'hold {sec(t_end - split)} s | click→picture {pic} s | 30 s after resume: dropped {drops}, late {lates} | ignored {ignored or "none"}')
### timeline2a.py
# timeline2a.py <run dir>: per 1 s from the [scrub] begin tick to land + 10 s — demux PCR range, DEMUX OUT per ES kind, tracer events.
import json, re, collections, sys, os
d = sys.argv[1]; log = open(os.path.join(d, 'marlin-media-tv.log'), errors='replace').read()
tb = int(re.findall(r'\[scrub\] begin at \d+ ms playing=\w+ tick=(\d+)', log)[-1]) * 1000
tl = int(re.findall(r'\[scrub\] lift target=\d+ ms tick=(\d+)', log)[-1]) * 1000
te = int(re.findall(r'\[scrub\] land from \d+ ms at \d+ ms tick=(\d+)', log)[-1]) * 1000
ev = []
for line in open(os.path.join(d, [f for f in os.listdir(d) if f.startswith('vlc-trace-')][0]), errors='replace'):
    try: o = json.loads(line)
    except Exception: continue
    b = o['Body']; b['_t'] = int(o['Timestamp']); ev.append(b)
ev.sort(key=lambda b: b['_t'])
print(f'lift at +{(tl-tb)/1e9:.2f} s, land at +{(te-tb)/1e9:.2f} s from begin')
B = collections.OrderedDict()
for b in ev:
    if b['_t'] < tb - 1e9 or b['_t'] > te + 10e9: continue
    k = int((b['_t'] - tb) // 1e9)
    x = B.setdefault(k, {'pcr': [], 'out': collections.Counter(), 'ev': collections.Counter()})
    if b.get('type') == 'DEMUX' and b.get('id') == 'PCR': x['pcr'].append(int(b['pcr']) // 1000)
    if b.get('type') == 'DEMUX' and b.get('stream') == 'OUT': x['out'][str(b.get('id')).split('/')[0]] += 1
    if b.get('event'): x['ev'][f"{b.get('id')}:{b['event']}"] += 1
for k, x in B.items():
    pr = f"{x['pcr'][0]}..{x['pcr'][-1]} ({len(x['pcr'])})" if x['pcr'] else '-'
    print(f"+{k:3d}s PCR {pr:28s} out {dict(x['out'])} events {dict(x['ev'])}")
### laps2a.py
# laps2a.py <run dir>: after the last [scrub] seek — input 'reset_user' tracer events (VLC's rebuffer laps), the demux PCR read in
# each, the last PCR before land against the target, and the first realtime video renders after land (pts, drift).
import json, re, sys, os
d = sys.argv[1]; log = open(os.path.join(d, 'marlin-media-tv.log'), errors='replace').read()
tb = int(re.findall(r'\[scrub\] begin at \d+ ms playing=\w+ tick=(\d+)', log)[-1]) * 1000
seeks = [(int(t) * 1000, int(ms)) for ms, t in re.findall(r'\[scrub\] seek (\d+) ms tick=(\d+)', log)]
tl = int(re.findall(r'\[scrub\] lift target=\d+ ms tick=(\d+)', log)[-1]) * 1000
te = int(re.findall(r'\[scrub\] land from \d+ ms at \d+ ms tick=(\d+)', log)[-1]) * 1000
ev = []
for line in open(os.path.join(d, [f for f in os.listdir(d) if f.startswith('vlc-trace-')][0]), errors='replace'):
    try: o = json.loads(line)
    except Exception: continue
    b = o['Body']; b['_t'] = int(o['Timestamp']); ev.append(b)
ev.sort(key=lambda b: b['_t'])
ls = seeks[-1][0]
print(f'last seek {seeks[-1][1]} ms at +{(ls - tb)/1e9:.3f} s; lift +{(tl-tb)/1e9:.3f} s; land +{(te-tb)/1e9:.3f} s')
laps = [b for b in ev if ls <= b['_t'] <= te and b.get('id') == 'input' and b.get('event') == 'reset_user']
print(f'input reset_user events after the last seek: {len(laps)}')
pcrs = [(b['_t'], int(b['pcr']) // 1000) for b in ev if b.get('type') == 'DEMUX' and b.get('id') == 'PCR' and ls <= b['_t'] <= te + 2e9]
for i, l in enumerate(laps):
    nxt = laps[i + 1]['_t'] if i + 1 < len(laps) else te
    w = [p for t, p in pcrs if l['_t'] <= t < nxt]
    print(f"  event {i+1:2d} +{(l['_t']-tb)/1e9:6.3f} s  PCR {w[0] if w else '-'} .. {w[-1] if w else '-'} µs/1000  ({len(w)} PCR)")
lr = max(t for t, p in pcrs if t <= te)
print(f'last demux PCR before land at +{(lr - tb)/1e9:.3f} s: {max(p for t, p in pcrs if t <= te)} (µs/1000); target {seeks[-1][1]} ms')
for b in [b for b in ev if b['_t'] > te and b.get('id') == 'video/1' and b.get('type') == 'RENDER' and b.get('mode') == 'realtime'][:3]:
    print(f"render after land +{(b['_t']-te)/1e9:.3f} s pts {int(b['pts'])//1000} (µs/1000) drift {int(b.get('drift',0))/1e6:.1f} ms")

cd "$REPO"

### probe2a.sh
#!/bin/zsh
# probe2a.sh — console launch with the default tracer path (fails, but libvlc prints the container path), attach
# Diag2aUITests/t0_Probe (Magicians: play 15 s, Page Up scripted drag, land), read the container path, copy the app log.
S=/private/tmp/claude-501/-Users-marlin1111-Xcode-Marlin-Media-TV/<UUID>/scratchpad/2a
BID=com.marlin1111.marlin-media-tv; DEV="Home Theater"
OUT=$S/probe; [ -e $OUT ] && { echo "$OUT exists — stop" >&2; exit 5; }; mkdir -p $OUT
cd ~/Xcode/"Marlin Media TV"
perl -e 'alarm 200; exec @ARGV' -- xcrun devicectl device process launch --console --terminate-existing --device "$DEV" $BID -VLCParams '("--no-color","--no-osd","--no-video-title-show","--no-snapshot-preview","--http-reconnect","--text-renderer=freetype","--avi-index=3","--audio-resampler=soxr","--tracer=json","--verbose=2")' > $OUT/console.out 2>&1 &
CPID=$!
/bin/sleep 12
xcodebuild test-without-building -project "Marlin Media TV.xcodeproj" -scheme "Marlin Media TV" -destination "platform=tvOS,name=$DEV" -derivedDataPath build/DerivedData -only-testing:"Marlin Media TVUITests/Diag2aUITests/t0_Probe" -resultBundlePath $OUT/r.xcresult > $OUT/xcodebuild.out 2>&1
echo "test exit $?"
grep -E 'Test Case.*(passed|failed)|error:' $OUT/xcodebuild.out | tail -3
xcrun xcresulttool export attachments --path $OUT/r.xcresult --output-path $OUT/att > /dev/null 2>&1
xcrun devicectl device copy from --device "$DEV" --domain-type appDataContainer --domain-identifier $BID --source Library/Caches/marlin-media-tv.log --destination $OUT/marlin-media-tv.log > /dev/null 2>&1 || echo "log copy failed"
kill $CPID 2>/dev/null; wait $CPID 2>/dev/null
C=$(grep -ao "opening logfile \`[^']*" $OUT/console.out | head -1 | sed -E 's/opening logfile `//; s#/Library/Logs/vlc-log.json##')
[ -n "$C" ] && { printf '%s' "$C" > $S/container.txt; echo "container path read: $(echo $C | sed -E 's/[0-9A-F]{8}-[0-9A-F-]{27}/<UUID>/')"; } || echo "no container path in console output"
grep -aE "\[harness\]|\[scrub\]" $OUT/marlin-media-tv.log | tail -20
