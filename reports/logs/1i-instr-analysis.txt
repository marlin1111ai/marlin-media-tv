### flagcheck.py
# flagcheck.py <app log>: last player session of an instrumented (1h lines) run. Timeline of every write of
# next_frame_need_data as the input sees it (setter old -> new, clears by control), with the nearest preceding app line;
# decoder notify counts per segment between app lines; and check1 tuple totals per pause segment
# (from each "[player] pause" / "state Paused" to the next "[player] play").
import re, sys, collections
lines = open(sys.argv[1], errors='replace').read().split('\n')
st = [i for i, l in enumerate(lines) if '[player] play() called' in l]
lines = lines[st[-1]:] if st else lines
app = re.compile(r'^\d+:\d\d:\d\d\.\d\d\d.*\[(player|framestep)\]')
last_app = ''
notify = collections.Counter()
seg = None; segs = []
for i, l in enumerate(lines):
    if app.match(l):
        if notify:
            print(f'      decoder notifies since: {dict(notify)}'); notify.clear()
        last_app = l.strip()[:100]
        if '[player] pause (' in l: seg = {'start': last_app, 'check1': collections.Counter()}; segs.append(seg)
        if '[player] play (' in l: seg = None
        if '[framestep]' in l or '[player] pause (' in l or '[player] play (' in l: print(f'  app: {last_app}')
        continue
    m = re.search(r'1h: INPUT_CONTROL_NEED_DATA_FRAME_NEXT (\d) -> (\d) now=(\d+)', l)
    if m: print(f'    setter {m.group(1)} -> {m.group(2)} now={m.group(3)}')
    m = re.search(r'1h: next_frame_need_data cleared by control (\d+) now=(\d+)', l)
    if m: print(f'    cleared by control {m.group(1)} now={m.group(2)}')
    m = re.search(r'1h: decoder notify need_data=(\d) frames_countdown=(-?\d+)', l)
    if m: notify[f'need_data={m.group(1)} countdown={m.group(2)}'] += 1
    m = re.search(r'1h: check1 .* es_out_GetBuffering=(\d) b_eof=(\d) next_frame_need_data=(\d) -> demux_allowed=(\d)', l)
    if m and seg is not None: seg['check1'][m.group(1, 2, 3, 4)] += 1
if notify: print(f'      decoder notifies since: {dict(notify)}')
for s in segs:
    print(f'pause segment from [{s["start"]}]: check1 (GetBuffering, b_eof, need_data, demux_allowed) -> {dict(s["check1"])}')
### pausereads.py
# pausereads.py <trace.json> [video id]: last session. Between the last vout 'paused' event and the next 'resumed':
# per 5 s bucket, the demux PCR range and DEMUX OUT counts per stream; then the totals, the highest PCR read while
# paused, the last PCR before the pause and the read-ahead (highest PCR while paused − last PCR before the pause).
# The input demuxes while paused only when es_out is buffering or next_frame_need_data is set (input.c:668–669), so a
# bucket with reads and no step in it marks the flag.
import json, sys, collections
vid = sys.argv[2] if len(sys.argv) > 2 else None
ev = []
for line in open(sys.argv[1], errors='replace'):
    try: o = json.loads(line)
    except Exception: continue
    b = o['Body']; b['_t'] = int(o['Timestamp']); ev.append(b)
ev.sort(key=lambda b: b['_t'])
if vid is None:
    vid = next((b['id'] for b in reversed(ev) if str(b.get('id', '')).startswith('video') and b.get('event') == 'paused'), 'video/1')
pz = max(i for i, b in enumerate(ev) if b.get('id') == vid and b.get('event') == 'paused')
rs = next((i for i in range(pz, len(ev)) if ev[i].get('id') == vid and ev[i].get('event') == 'resumed'), len(ev) - 1)
t0 = ev[pz]['_t']; tr = ev[rs]['_t']
pre = [int(b['pcr']) for b in ev[:pz] if b.get('type') == 'DEMUX' and b.get('id') == 'PCR']
print(f'{vid}: paused at trace t0; resumed at +{(tr - t0) / 1e9:.2f} s; last PCR before pause {pre[-1] // 1000 if pre else None} ms')
B = collections.OrderedDict(); tot = collections.Counter(); pcr_max = None
for b in ev[pz:rs]:
    k = int((b['_t'] - t0) // 5e9) * 5
    d = B.setdefault(k, {'pcr': [], 'out': collections.Counter()})
    if b.get('type') == 'DEMUX' and b.get('id') == 'PCR':
        p = int(b['pcr']); d['pcr'].append(p // 1000); pcr_max = p if pcr_max is None else max(pcr_max, p)
    if b.get('type') == 'DEMUX' and b.get('stream') == 'OUT':
        d['out'][str(b.get('id'))] += 1; tot[str(b.get('id'))] += 1
for k in range(0, int((tr - t0) / 1e9) + 5, 5):
    d = B.get(k, {'pcr': [], 'out': collections.Counter()})
    pr = f"{d['pcr'][0]}..{d['pcr'][-1]} ms ({len(d['pcr'])})" if d['pcr'] else '-'
    print(f'+{k:3d}s  PCR {pr:34s} demux out {dict(d["out"]) if d["out"] else "-"}')
print(f'totals while paused: demux out {dict(tot)}; highest PCR read {pcr_max // 1000 if pcr_max else None} ms; '
      f'read-ahead {(pcr_max - pre[-1]) / 1e6 if pcr_max and pre else 0:.3f} s')
