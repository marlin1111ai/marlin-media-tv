### scrub2d.py
# scrub2c.py <run dir>...: pass 2d numbers per scrub (several scrubs per session; clean recipe build, no instrumentation).
#   app log (from the last "[player] play() called"), in order per scrub:
#     [scrub] begin at S ms playing=B tick=T         position before the drag
#     [scrub] lift target=X ms tick=T                target shown when the thumb lifts
#     [scrub] land (select|playPause) from S ms at X ms tick=T   /  [scrub] cancel (menu) at X ms, back to S ms playing=B tick=T
#     [scrub] seek X ms after play tick=T            the one seek (landing only)
#     [scrub] land|cancel +3 s time=N ms             position 3 s after the ending
#   tick = CLOCK_MONOTONIC µs = VLC's vlc_tick_now(); tracer Timestamp = ns on the same clock.
#   reads while paused    = DEMUX OUT events (every ES) from begin to the ending, with the demux PCR range; plus the
#                           tracer's input/decoder 'flushed' events in that window (a seek flushes; none expected).
#   picture               = first realtime video RENDER after the ending whose pts is within 2 s of the target
#                           (landing) or of the start (cancel from playing). click → picture = its time − ending tick.
#   landing offset        = that picture's pts − target.   +3 s error = time(+3 s) − (target + (3 s − click → picture)).
#   old pictures          = realtime video RENDERs between the ending and that picture whose pts is not near the target.
#   dropped / late        = video RENDER 'toolate' / 'late' from the picture to +30 s (or the next begin, if sooner).
import json, re, sys, os

def ms(ns): return round(ns / 1e6, 1)

for d in sys.argv[1:]:
    tag = os.path.basename(d.rstrip('/'))
    log = open(os.path.join(d, 'marlin-media-tv.log'), errors='replace').read().split('\n')
    st = [i for i, l in enumerate(log) if '[player] play() called' in l]; log = log[st[-1]:] if st else log
    tr = [f for f in os.listdir(d) if f.startswith('vlc-trace-')]
    ev = []
    for line in open(os.path.join(d, tr[0]), errors='replace'):
        try: o = json.loads(line)
        except Exception: continue
        b = o['Body']; b['_t'] = int(o['Timestamp']); ev.append(b)
    ev.sort(key=lambda x: x['_t'])
    vid = 'video/1'
    scrubs, cur = [], None
    for l in log:
        m = re.search(r'\[scrub\] begin at (\d+) ms (?:playing=(\w+) )?tick=(\d+)', l)
        if m: cur = {'start': int(m.group(1)), 'playing': m.group(2) == 'true', 'tb': int(m.group(3)) * 1000}; scrubs.append(cur); continue
        if cur is None: continue
        m = re.search(r'\[scrub\] lift target=(\d+) ms tick=(\d+)', l)
        if m: cur['lift'] = int(m.group(1)); continue
        m = re.search(r'\[scrub\] land (\w+) from (\d+) ms at (\d+) ms tick=(\d+)', l)
        if m: cur.update(kind='land ' + m.group(1), target=int(m.group(3)), te=int(m.group(4)) * 1000); continue
        m = re.search(r'\[scrub\] cancel (\w+) at (\d+) ms, back to (\d+) ms (?:playing=(\w+) )?tick=(\d+)', l)
        if m: cur.update(kind='cancel ' + m.group(1), target=int(m.group(2)), te=int(m.group(5)) * 1000); continue
        m = re.search(r'\[scrub\] seek (\d+) ms after play tick=(\d+)', l)
        if m: cur['seek'] = (int(m.group(1)), int(m.group(2)) * 1000); continue
        m = re.search(r'\[scrub\] (?:land|cancel) \+3 s time=(\d+) ms', l)
        if m: cur['t3'] = int(m.group(1)); continue
        if 'ignored while scrubbing' in l or 'drag not taken' in l: cur.setdefault('ignored', []).append(l.split('[scrub] ')[1])
    print(f'== {tag}: {len(scrubs)} scrubs')
    for n, s in enumerate(scrubs, 1):
        if 'te' not in s: print(f'  {n}: begin at {s["start"]} ms, no ending line'); continue
        nxt = scrubs[n]['tb'] if n < len(scrubs) else ev[-1]['_t']
        w = [x for x in ev if s['tb'] <= x['_t'] <= s['te']]
        reads = sum(1 for x in w if x.get('type') == 'DEMUX' and x.get('stream') == 'OUT')
        pcr = [int(x['pcr']) // 1000 for x in w if x.get('type') == 'DEMUX' and x.get('id') == 'PCR']
        flushed = sum(1 for x in w if x.get('event') == 'flushed')
        land = s['kind'].startswith('land')
        aim = s['target'] if land else s['start']
        after = [x for x in ev if s['te'] <= x['_t'] < nxt and x.get('id') == vid and x.get('type') == 'RENDER' and x.get('mode') == 'realtime']
        pic = next((x for x in after if abs(int(x['pts']) // 1000 - aim * 1000) <= 2_000_000), None)
        old = sum(1 for x in after if pic is not None and x['_t'] < pic['_t'])
        c2p = off = err3 = drops = lates = None
        if pic is not None:
            c2p = round((pic['_t'] - s['te']) / 1e9, 3)
            off = round((int(pic['pts']) // 1000 - aim * 1000) / 1000)   # ms
            end30 = min(pic['_t'] + 30e9, nxt)
            win = [x for x in ev if pic['_t'] <= x['_t'] <= end30 and x.get('id') == vid]
            drops = sum(1 for x in win if x.get('event') == 'toolate'); lates = sum(1 for x in win if x.get('event') == 'late')
            if s.get('t3') is not None and (land or s['playing']):
                err3 = s['t3'] - (aim + max(0, 3000 - int(c2p * 1000)))
        if not land and not s['playing'] and s.get('t3') is not None:
            err3 = s['t3'] - aim
        # trace position at +3 s: pts of the last realtime video render at or before ending + 3.0 s (VLCKit's time can lag)
        at3 = [x for x in after if x['_t'] <= s['te'] + 3e9]
        pts3 = int(at3[-1]['pts']) // 1000000 if at3 else None
        s['pts3'] = pts3
        s['pts3err'] = (pts3 - (aim + max(0, 3000 - int(c2p * 1000)))) if (pts3 is not None and c2p is not None and (land or s['playing'])) else ((pts3 - aim) if pts3 is not None else None)
        if os.environ.get('MD'):
            ending = {'land select': 'click', 'land playPause': 'Play/Pause', 'cancel menu': 'Menu'}[s['kind']]
            direction = 'fwd' if s.get('lift', 0) > s['start'] else 'back'
            print(f"| {tag} | {n} | {'playing' if s['playing'] else 'paused'} | {direction} | {ending} | {s['start']} | {s.get('lift')} | "
                  f"{s.get('pts3')} ({s.get('pts3err'):+d}) | {s.get('t3')} | {off:+d} | {c2p} | {drops} / {lates} | {reads} |")
            continue
        print(f"  {n}: {s['kind']:15s} from {'playing' if s['playing'] else 'paused '} | before {s['start']} ms | target shown {s.get('lift')} ms | "
              f"ending at {s['target']} ms | seek {s.get('seek', (None,))[0]} | +3 s {s.get('t3')} ms (error {err3} ms), trace pts at +3 s {s.get('pts3')} ms (error {s.get('pts3err')} ms) | "
              f"picture pts offset {off} ms, click→picture {c2p} s, old pictures first {old} | dropped {drops}, late {lates} | "
              f"reads while paused {reads} (PCR {(pcr[0], pcr[-1]) if pcr else None}), flushed events {flushed} | ignored {s.get('ignored', 'none')}")

### ctl2d.py
# ctl2d.py <run dir>...: pass 2d control session checks from the app log (from the last "[player] play() called").
#   playing drags: each "[harness] scripted drag" while playing must be followed by "[scrub] drag while playing: no action",
#                  with no "[scrub] begin" and no "[player] state Paused" before the next harness action (the +30 s skip).
#   skips:         "[skip] ±N s (click) before= target=" and "after=".
#   frame steps:   "[framestep] ±1 go…Frame before=" and "after(400 ms)=" per click, with the step from the previous position.
import re, sys, os

for d in sys.argv[1:]:
    tag = os.path.basename(d.rstrip('/'))
    log = open(os.path.join(d, 'marlin-media-tv.log'), errors='replace').read().split('\n')

    print(f'== {tag}')

### fs2d.py
# fs2d.py <run dir>: pass 2d frame-step check (pass 1k's picdiff method). Screenshots p2d-<tag>-fs-00-paused,
# fs-01..05-L, fs-06..10-R from the xcresult manifest, scaled to 960×540 BMP with sips; mean absolute difference ×10
# (0-2550) over the video area above the overlay (x 130..830, y 120..380), excluding the pause glyph (x 440..520, y 145..220).
# One picture per click: each click differs from the one before; the right clicks retrace the left ones —
# R6 = L4, R7 = L3, R8 = L2, R9 = L1, R10 = paused (MAD 0). Also the app's "[framestep] … after(400 ms)" positions.
import json, os, re, struct, subprocess, sys
d = sys.argv[1]
tmp = os.path.join(d, 'fsbmp'); os.makedirs(tmp, exist_ok=True)
shots = {}
for t in json.load(open(os.path.join(d, 'att', 'manifest.json'))):
    for a in t.get('attachments', []):
        n = a.get('suggestedHumanReadableName', '')
        m = re.search(r'-fs-(\d\d)-(\w+)_0_', n)
        if m: shots[int(m.group(1))] = (m.group(2), os.path.join(d, 'att', a['exportedFileName']))
px = {}
for k, (label, src) in sorted(shots.items()):
    bmp = os.path.join(tmp, f'{k:02d}.bmp')
    subprocess.run(['sips', '-s', 'format', 'bmp', '-z', '540', '960', src, '--out', bmp], capture_output=True)
    data = open(bmp, 'rb').read()
    off, = struct.unpack_from('<I', data, 10); w, h = struct.unpack_from('<ii', data, 18); bpp, = struct.unpack_from('<H', data, 28)
    step = bpp // 8; row = (w * step + 3) & ~3
    def at(x, y):
        yy = (h - 1 - y) if h > 0 else y
        i = off + yy * row + x * step
        return data[i] + data[i + 1] + data[i + 2]
    px[k] = [at(x, y) for y in range(120, 380, 2) for x in range(130, 830, 2) if not (440 <= x < 520 and 145 <= y < 220)]
def mad(a, b): return round(10 * sum(abs(p - q) for p, q in zip(px[a], px[b])) / (3 * len(px[a])))
names = {k: (f'{k:02d}-{v[0]}') for k, v in shots.items()}
print('MAD ×10 (0 = identical picture area)')
print('         ' + ' '.join(f'{names[k][:6]:>6}' for k in sorted(px)))
for a in sorted(px):
    print(f'{names[a]:>8} ' + ' '.join(f'{mad(a, b):6d}' for b in sorted(px)))
steps_differ = all(mad(k, k - 1) > 0 for k in range(1, 11) if k in px and k - 1 in px)
retrace = {6: 4, 7: 3, 8: 2, 9: 1, 10: 0}
retrace_ok = all(mad(r, l) == 0 for r, l in retrace.items() if r in px and l in px)
print(f'each click differs from the previous picture: {steps_differ} ({[mad(k, k - 1) for k in range(1, 11) if k in px and k - 1 in px]})')
print(f'right clicks retrace the left clicks (R6=L4 … R10=paused, MAD 0): {retrace_ok} ({[(r, l, mad(r, l)) for r, l in retrace.items() if r in px and l in px]})')
log = open(os.path.join(d, 'marlin-media-tv.log'), errors='replace').read()
print('app positions:', re.findall(r'\[framestep\] ([−+]1) after\(400 ms\)=(\d+) ms', log))

#!/bin/zsh
# probe2d.sh — console launch with the default tracer path (fails, but libvlc prints the container path), attach
# Diag2dUITests/t0_Probe (Magicians: play 15 s, Page Up scripted drag, land), read the container path, copy the app log.
S=/private/tmp/claude-501/-Users-marlin1111-Xcode-Marlin-Media-TV/<UUID>/scratchpad/2d
BID=com.marlin1111.marlin-media-tv; DEV="Home Theater"
OUT=$S/probe; [ -e $OUT ] && { echo "$OUT exists — stop" >&2; exit 5; }; mkdir -p $OUT
cd ~/Xcode/"Marlin Media TV"
perl -e 'alarm 200; exec @ARGV' -- xcrun devicectl device process launch --console --terminate-existing --device "$DEV" $BID -VLCParams '("--no-color","--no-osd","--no-video-title-show","--no-snapshot-preview","--http-reconnect","--text-renderer=freetype","--avi-index=3","--audio-resampler=soxr","--tracer=json","--verbose=2")' > $OUT/console.out 2>&1 &
CPID=$!
/bin/sleep 12
xcodebuild test-without-building -project "Marlin Media TV.xcodeproj" -scheme "Marlin Media TV" -destination "platform=tvOS,name=$DEV" -derivedDataPath build/DerivedData -only-testing:"Marlin Media TVUITests/Diag2dUITests/t0_Probe" -resultBundlePath $OUT/r.xcresult > $OUT/xcodebuild.out 2>&1
echo "test exit $?"
grep -E 'Test Case.*(passed|failed)|error:' $OUT/xcodebuild.out | tail -3
xcrun xcresulttool export attachments --path $OUT/r.xcresult --output-path $OUT/att > /dev/null 2>&1
xcrun devicectl device copy from --device "$DEV" --domain-type appDataContainer --domain-identifier $BID --source Library/Caches/marlin-media-tv.log --destination $OUT/marlin-media-tv.log > /dev/null 2>&1 || echo "log copy failed"
kill $CPID 2>/dev/null; wait $CPID 2>/dev/null
C=$(grep -ao "opening logfile \`[^']*" $OUT/console.out | head -1 | sed -E 's/opening logfile `//; s#/Library/Logs/vlc-log.json##')
[ -n "$C" ] && { printf '%s' "$C" > $S/container.txt; echo "container path read: $(echo $C | sed -E 's/[0-9A-F]{8}-[0-9A-F-]{27}/<UUID>/')"; } || echo "no container path in console output"
grep -aE "\[harness\]|\[scrub\]" $OUT/marlin-media-tv.log | tail -20
